   117→const clamp01 = (v: number) => Math.min(1, Math.max(0, v))
   118→
   119→/**
   120→ * Fade vinculado à posição de rolagem (sem transição CSS — o estilo é recalculado a cada frame):
   121→ * - rolando para baixo: o que entra por baixo sofre fade in subindo (+44px → 0) e o que
   122→ *   sai por cima sofre fade out (0 → -30px);
   123→ * - rolando para cima (reverse): o mesmo cálculo posicional inverte o efeito — o conteúdo
   124→ *   superior volta com fade in descendo e o inferior some com fade out.
   125→ */
   126→type FadeApi = { register: (el: HTMLDivElement | null) => void }
   127→const FadeCtx = createContext<FadeApi | null>(null)
   128→
   129→function useFadeController(): FadeApi {
   130→  const items = useRef(new Set<HTMLDivElement>())
   131→  const rafRef = useRef(0)
   132→
   133→  const update = useCallback(() => {
   134→    rafRef.current = 0
   135→    const H = window.innerHeight || 1
   136→    const reduced = window.matchMedia('(prefers-reduced-motion: reduce)').matches
   137→    items.current.forEach((el) => {
   138→      if (!el.isConnected) {
   139→        items.current.delete(el)
   140→        return
   141→      }
   142→      if (reduced) {
   143→        el.style.opacity = '1'
   144→        el.style.transform = ''
   145→        return
   146→      }
   147→      const rect = el.getBoundingClientRect()
   148→      // Desconta o transform aplicado no frame anterior para medir a posição de layout
   149→      const prevY = Number(el.dataset.fadeY ?? 0)
   150→      const top = rect.top - prevY
   151→      const bottom = rect.bottom - prevY
   152→      // Entrada: topo percorrendo os 22% inferiores da viewport
   153→      const tIn = clamp01((H - top) / (H * 0.22))
   154→      // Saída: base adentrando os 28% superiores da viewport
   155→      const tOut = clamp01(bottom / (H * 0.28))
   156→      // Blocos ancorados na primeira dobra nascem visíveis e sofrem apenas o fade de saída
   157→      const anchored = el.dataset.fadeAnchored === '1'
   158→      const opacity = anchored ? tOut : Math.min(tIn, tOut)
   159→      const y = (anchored ? 0 : (1 - tIn) * 44) - (1 - tOut) * 30
   160→      el.dataset.fadeY = y.toFixed(1)
   161→      el.style.opacity = opacity.toFixed(3)
   162→      el.style.transform = `translate3d(0, ${y.toFixed(1)}px, 0)`
   163→    })
   164→  }, [])
   165→
   166→  const schedule = useCallback(() => {
   167→    if (!rafRef.current) rafRef.current = requestAnimationFrame(update)
   168→  }, [update])
   169→
   170→  const register = useCallback((el: HTMLDivElement | null) => {
   171→    if (el) items.current.add(el)
   172→  }, [])
   173→
   174→  useEffect(() => {
   175→    schedule()
   176→    window.addEventListener('scroll', schedule, { passive: true })
   177→    window.addEventListener('resize', schedule)
   178→    // Recalcula após fontes/imagens estabilizarem o layout
   179→    const t1 = window.setTimeout(schedule, 350)
   180→    const t2 = window.setTimeout(schedule, 900)
   181→    document.fonts?.ready.then(schedule).catch(() => {})
   182→    return () => {
   183→      window.removeEventListener('scroll', schedule)
   184→      window.removeEventListener('resize', schedule)
   185→      window.clearTimeout(t1)
   186→      window.clearTimeout(t2)
   187→      if (rafRef.current) cancelAnimationFrame(rafRef.current)
   188→    }
   189→  }, [schedule])
   190→
   191→  return { register }
   192→}

   194→/** Bloco que participa do fade de rolagem; startVisible mantém o bloco visível antes da hidratação e anchored desativa o fade de entrada (só sai por cima) */
   195→function ScrollFade({ children, className, startVisible, anchored }: {
   196→  children: React.ReactNode
   197→  className?: string
   198→  startVisible?: boolean
   199→  anchored?: boolean
   200→}) {
   201→  const fade = useContext(FadeCtx)
   202→  return (
   203→    <div
   204→      ref={fade?.register}
   205→      className={cn('relative', className)}
   206→      data-fade-anchored={anchored ? '1' : undefined}
   207→      data-fade-y={startVisible ? undefined : '44'}
   208→      style={{
   209→        opacity: startVisible ? undefined : 0,
   210→        transform: startVisible ? undefined : 'translate3d(0, 44px, 0)',
   211→        willChange: 'opacity, transform',
   212→      }}
   213→    >
   214→      {children}
   215→    </div>
   216→  )
   217→}
