    42→function readRememberEmail(): string | null {
    43→  try {
    44→    const raw = window.localStorage.getItem(REMEMBER_KEY)
    45→    if (!raw) return null
    46→    const email = decodeURIComponent(atob(raw))
    47→    return email.includes('@') ? email : null
    48→  } catch {
    49→    return null
    50→  }
    51→}

   142→const clamp01 = (v: number) => Math.min(1, Math.max(0, v))
   143→
   144→/**
   145→ * Fade vinculado à posição de rolagem (sem transição CSS — o estilo é recalculado a cada frame):
   146→ * - rolando para baixo: o que entra por baixo sofre fade in subindo (+44px → 0) e o que
   147→ *   sai por cima sofre fade out (0 → -30px);
   148→ * - rolando para cima (reverse): o mesmo cálculo posicional inverte o efeito — o conteúdo
   149→ *   superior volta com fade in descendo e o inferior some com fade out.
   150→ */
   151→type FadeApi = { register: (el: HTMLDivElement | null) => void }
   152→const FadeCtx = createContext<FadeApi | null>(null)
   153→
   154→function useFadeController(): FadeApi {
   155→  const items = useRef(new Set<HTMLDivElement>())
   156→  const rafRef = useRef(0)
   157→
   158→  const update = useCallback(() => {
   159→    rafRef.current = 0
   160→    const H = window.innerHeight || 1
   161→    const reduced = window.matchMedia('(prefers-reduced-motion: reduce)').matches
   162→    items.current.forEach((el) => {
   163→      if (!el.isConnected) {
   164→        items.current.delete(el)
   165→        return
   166→      }
   167→      if (reduced) {
   168→        el.style.opacity = '1'
   169→        el.style.transform = ''
   170→        return
   171→      }
   172→      const rect = el.getBoundingClientRect()
   173→      // Desconta o transform aplicado no frame anterior para medir a posição de layout
   174→      const prevY = Number(el.dataset.fadeY ?? 0)
   175→      const top = rect.top - prevY
   176→      const bottom = rect.bottom - prevY
   177→      // Entrada: topo percorrendo os 22% inferiores da viewport
   178→      const tIn = clamp01((H - top) / (H * 0.22))
   179→      // Saída: base adentrando os 28% superiores da viewport
   180→      const tOut = clamp01(bottom / (H * 0.28))
   181→      // Blocos ancorados na primeira dobra nascem visíveis e sofrem apenas o fade de saída
   182→      const anchored = el.dataset.fadeAnchored === '1'
   183→      const opacity = anchored ? tOut : Math.min(tIn, tOut)
   184→      const y = (anchored ? 0 : (1 - tIn) * 44) - (1 - tOut) * 30
   185→      el.dataset.fadeY = y.toFixed(1)
   186→      el.style.opacity = opacity.toFixed(3)
   187→      el.style.transform = `translate3d(0, ${y.toFixed(1)}px, 0)`
   188→    })
   189→  }, [])
   190→
   191→  const schedule = useCallback(() => {
   192→    if (!rafRef.current) rafRef.current = requestAnimationFrame(update)
   193→  }, [update])
   194→
   195→  const register = useCallback((el: HTMLDivElement | null) => {
   196→    if (el) items.current.add(el)
   197→  }, [])
   198→
   199→  useEffect(() => {
   200→    schedule()
   201→    window.addEventListener('scroll', schedule, { passive: true })
   202→    window.addEventListener('resize', schedule)
   203→    // Recalcula após fontes/imagens estabilizarem o layout
   204→    const t1 = window.setTimeout(schedule, 350)
   205→    const t2 = window.setTimeout(schedule, 900)
   206→    document.fonts?.ready.then(schedule).catch(() => {})
   207→    return () => {
   208→      window.removeEventListener('scroll', schedule)
   209→      window.removeEventListener('resize', schedule)
   210→      window.clearTimeout(t1)
   211→      window.clearTimeout(t2)
   212→      if (rafRef.current) cancelAnimationFrame(rafRef.current)
   213→    }
   214→  }, [schedule])
   215→
   216→  return { register }
   217→}

   219→/** Bloco que participa do fade de rolagem; startVisible mantém o bloco visível antes da hidratação e anchored desativa o fade de entrada (só sai por cima) */
   220→function ScrollFade({ children, className, startVisible, anchored }: {
   221→  children: React.ReactNode
   222→  className?: string
   223→  startVisible?: boolean
   224→  anchored?: boolean
   225→}) {
   226→  const fade = useContext(FadeCtx)
   227→  return (
   228→    <div
   229→      ref={fade?.register}
   230→      className={cn('relative', className)}
   231→      data-fade-anchored={anchored ? '1' : undefined}
   232→      data-fade-y={startVisible ? undefined : '44'}
   233→      style={{
   234→        opacity: startVisible ? undefined : 0,
   235→        transform: startVisible ? undefined : 'translate3d(0, 44px, 0)',
   236→        willChange: 'opacity, transform',
   237→      }}
   238→    >
   239→      {children}
   240→    </div>
   241→  )
   242→}
