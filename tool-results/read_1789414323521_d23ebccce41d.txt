    85→/**
    86→ * Contador animado do total acumulado — conta de R$ 0 até o valor atual na entrada
    87→ * e sobe/desce como um odômetro sempre que a soma inteligente da comanda muda
    88→ * (novos itens somam; itens removidos pelo garçom subtraem).
    89→ */
    90→function useAnimatedMoney(target: number, duration = 900): number {
    91→  const [value, setValue] = useState(0)
    92→  const fromRef = useRef(0)
    93→
    94→  useEffect(() => {
    95→    const from = fromRef.current
    96→    // Diferença irrelevante → conclui em um frame, sem cascata de renders
    97→    const dur = Math.abs(target - from) < 0.004 ? 0 : duration
    98→    let raf = 0
    99→    const start = performance.now()
   100→    const step = (now: number) => {
   101→      const t = dur === 0 ? 1 : Math.min(1, (now - start) / dur)
   102→      const eased = 1 - Math.pow(1 - t, 3) // easeOutCubic — desacelera no fim, estilo caixa registradora
   103→      const v = from + (target - from) * eased
   104→      fromRef.current = v
   105→      setValue(v)
   106→      if (t < 1) raf = requestAnimationFrame(step)
   107→      else {
   108→        fromRef.current = target
   109→        setValue(target)
   110→      }
   111→    }
   112→    raf = requestAnimationFrame(step)
   113→    return () => cancelAnimationFrame(raf)
   114→  }, [target, duration])
   115→
   116→  return value
   117→}
