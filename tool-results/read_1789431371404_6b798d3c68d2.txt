    88→    88→  if (!iso) return '—'

    98→    98→function deadlineLabel(e: EstablishmentRow): { text: string; tone: 'ok' | 'warn' | 'bad' | 'muted' } {
    99→    99→  if (e.billingStatus === 'CANCELED') return { text: 'Cancelado', tone: 'muted' }
   100→   100→  if (e.billingStatus === 'TRIAL') {
   101→   101→    const d = daysFromNow(e.trialEndsAt)
   102→   102→    if (!e.trialEndsAt) return { text: 'Sem prazo de teste', tone: 'muted' }
   103→   103→    if (d > 0) return { text: `Teste até ${fmtDate(e.trialEndsAt)} · ${d}d`, tone: d <= 3 ? 'warn' : 'ok' }
   104→   104→    return { text: `Teste expirou em ${fmtDate(e.trialEndsAt)}`, tone: 'bad' }
   105→   105→  }
   106→   106→  const d = daysFromNow(e.currentPeriodEnd)
   107→   107→  if (!e.currentPeriodEnd) return { text: 'Sem vencimento', tone: 'muted' }
   108→   108→  if (d > 0) return { text: `Vence ${fmtDate(e.currentPeriodEnd)} · ${d}d`, tone: d <= 5 ? 'warn' : 'ok' }
   109→   109→  return { text: `Venceu ${fmtDate(e.currentPeriodEnd)}`, tone: 'bad' }
   110→   110→}

   332→   332→function PermissionsDialog({ est, onClose }: { est: EstablishmentRow | null; onClose: () => void }) {
   333→   333→  const queryClient = useQueryClient()
   334→   334→  const [matrix, setMatrix] = useState<Record<string, string[]>>({})
   335→   335→  const [loadedFor, setLoadedFor] = useState<string | null>(null)
   336→   336→  if (est && loadedFor !== est.id) {
   337→   337→    setLoadedFor(est.id)
   338→   338→    // Efetivo = padrão + overrides salvos
   339→   339→    let overrides: Record<string, string[]> = {}
   340→   340→    try { overrides = JSON.parse(est.permissions || '{}') } catch { /* ignora */ }
   341→   341→    const effective: Record<string, string[]> = {}
   342→   342→    for (const v of MATRIX_VIEWS) {
   343→   343→      effective[v.key] = overrides[v.key] ?? DEFAULT_VIEW_ROLES[v.key]
   344→   344→    }
   345→   345→    setMatrix(effective)
   346→   346→  }
   347→   347→
   348→   348→  const save = useMutation({
   349→   349→    mutationFn: (permissions: Record<string, string[]>) =>
   350→   350→      apiPatch(`/api/platform/establishments/${est!.id}`, { permissions }),
   351→   351→    onSuccess: () => {
   352→   352→      toast.success('Permissões atualizadas — sessões dos usuários aplicam no próximo carregamento')
   353→   353→      void queryClient.invalidateQueries({ queryKey: ['platform'] })
   354→   354→      onClose()
   355→   355→    },
   356→   356→    onError: (e: Error) => toast.error(e.message),
   357→   357→  })
   358→   358→
   359→   359→  const toggle = (view: string, role: string) => {
   360→   360→    setMatrix((m) => {
   361→   361→      const cur = m[view] ?? []
   362→   362→      return { ...m, [view]: cur.includes(role) ? cur.filter((r) => r !== role) : [...cur, role] }
   363→   363→    })
   364→   364→  }
   365→   365→
   366→   366→  const customized = est ? hasPermissionOverrides(est.permissions) : false
   367→   367→
   368→   368→  return (
   369→   369→    <Dialog open={!!est} onOpenChange={(v) => !v && onClose()}>
   370→   370→      <DialogContent className="sm:max-w-2xl">
   371→   371→        <DialogHeader>
   372→   372→          <DialogTitle className="flex items-center gap-2">
   373→   373→            <ShieldCheck className="h-4 w-4 text-primary" /> Permissões · {est?.name}
   374→   374→            {customized && <Badge variant="outline" className="text-[10px] border-primary/40 text-primary">Customizado</Badge>}
   375→   375→          </DialogTitle>
   376→   376→          <DialogDescription>
   377→   377→            Define quais cargos acessam cada tela deste estabelecimento. Deixe igual ao padrão para herdar a configuração base.
   378→   378→            Desenvolvedor CEO, Administrador e Gerente têm acesso irrestrito a todas as telas.
   379→   379→          </DialogDescription>
   380→   380→        </DialogHeader>
   381→   381→        <div className="overflow-x-auto max-h-[52vh] overflow-y-auto">
   382→   382→          <Table>
   383→   383→            <TableHeader>
   384→   384→              <TableRow>
   385→   385→                <TableHead className="pl-1">Tela</TableHead>
   386→   386→                {TENANT_ROLE_LIST.map((r) => (
   387→   387→                  <TableHead key={r} className="text-center text-[11px]">{TENANT_ROLE_LABELS[r]}</TableHead>
   388→   388→                ))}
   389→   389→              </TableRow>
   390→   390→            </TableHeader>
   391→   391→            <TableBody>
