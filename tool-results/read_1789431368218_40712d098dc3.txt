    87→function fmtDate(iso: string | null | undefined): string {
    88→  if (!iso) return '—'
    89→  return new Date(iso).toLocaleDateString('pt-BR', { day: '2-digit', month: '2-digit', year: '2-digit' })
    90→}
    91→
    92→function daysFromNow(iso: string | null | undefined): number {
    93→  if (!iso) return 0
    94→  return Math.ceil((new Date(iso).getTime() - Date.now()) / day)
    95→}

    97→/** Prazo amigável por status de cobrança */
    98→function deadlineLabel(e: EstablishmentRow): { text: string; tone: 'ok' | 'warn' | 'bad' | 'muted' } {
    99→  if (e.billingStatus === 'CANCELED') return { text: 'Cancelado', tone: 'muted' }
   100→  if (e.billingStatus === 'TRIAL') {
   101→    const d = daysFromNow(e.trialEndsAt)
   102→    if (!e.trialEndsAt) return { text: 'Sem prazo de teste', tone: 'muted' }
   103→    if (d > 0) return { text: `Teste até ${fmtDate(e.trialEndsAt)} · ${d}d`, tone: d <= 3 ? 'warn' : 'ok' }
   104→    return { text: `Teste expirou em ${fmtDate(e.trialEndsAt)}`, tone: 'bad' }
   105→  }
   106→  const d = daysFromNow(e.currentPeriodEnd)
   107→  if (!e.currentPeriodEnd) return { text: 'Sem vencimento', tone: 'muted' }
   108→  if (d > 0) return { text: `Vence ${fmtDate(e.currentPeriodEnd)} · ${d}d`, tone: d <= 5 ? 'warn' : 'ok' }
   109→  return { text: `Venceu ${fmtDate(e.currentPeriodEnd)}`, tone: 'bad' }
   110→}

   136→function toInputDate(iso: string | null | undefined): string {
   137→  if (!iso) return ''
   138→  const d = new Date(iso)
   139→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`
   140→}

   331→/** ---------- Diálogo: permissões por tela × cargo ---------- */
   332→function PermissionsDialog({ est, onClose }: { est: EstablishmentRow | null; onClose: () => void }) {
   333→  const queryClient = useQueryClient()
   334→  const [matrix, setMatrix] = useState<Record<string, string[]>>({})
   335→  const [loadedFor, setLoadedFor] = useState<string | null>(null)
   336→  if (est && loadedFor !== est.id) {
   337→    setLoadedFor(est.id)
   338→    // Efetivo = padrão + overrides salvos
   339→    let overrides: Record<string, string[]> = {}
   340→    try { overrides = JSON.parse(est.permissions || '{}') } catch { /* ignora */ }
   341→    const effective: Record<string, string[]> = {}
   342→    for (const v of MATRIX_VIEWS) {
   343→      effective[v.key] = overrides[v.key] ?? DEFAULT_VIEW_ROLES[v.key]
   344→    }
   345→    setMatrix(effective)
   346→  }
   347→
   348→  const save = useMutation({
   349→    mutationFn: (permissions: Record<string, string[]>) =>
   350→      apiPatch(`/api/platform/establishments/${est!.id}`, { permissions }),
   351→    onSuccess: () => {
   352→      toast.success('Permissões atualizadas — sessões dos usuários aplicam no próximo carregamento')
   353→      void queryClient.invalidateQueries({ queryKey: ['platform'] })
   354→      onClose()
   355→    },
   356→    onError: (e: Error) => toast.error(e.message),
   357→  })
   358→
   359→  const toggle = (view: string, role: string) => {
   360→    setMatrix((m) => {
   361→      const cur = m[view] ?? []
   362→      return { ...m, [view]: cur.includes(role) ? cur.filter((r) => r !== role) : [...cur, role] }
   363→    })
   364→  }
   365→
   366→  const customized = est ? hasPermissionOverrides(est.permissions) : false
   367→
   368→  return (
   369→    <Dialog open={!!est} onOpenChange={(v) => !v && onClose()}>
   370→      <DialogContent className="sm:max-w-2xl">
   371→        <DialogHeader>
   372→          <DialogTitle className="flex items-center gap-2">
   373→            <ShieldCheck className="h-4 w-4 text-primary" /> Permissões · {est?.name}
   374→            {customized && <Badge variant="outline" className="text-[10px] border-primary/40 text-primary">Customizado</Badge>}
   375→          </DialogTitle>
   376→          <DialogDescription>
   377→            Define quais cargos acessam cada tela deste estabelecimento. Deixe igual ao padrão para herdar a configuração base.
   378→            Desenvolvedor CEO, Administrador e Gerente têm acesso irrestrito a todas as telas.
   379→          </DialogDescription>
   380→        </DialogHeader>
   381→        <div className="overflow-x-auto max-h-[52vh] overflow-y-auto">
   382→          <Table>
   383→            <TableHeader>
   384→              <TableRow>
   385→                <TableHead className="pl-1">Tela</TableHead>
   386→                {TENANT_ROLE_LIST.map((r) => (
   387→                  <TableHead key={r} className="text-center text-[11px]">{TENANT_ROLE_LABELS[r]}</TableHead>
   388→                ))}
   389→              </TableRow>
   390→            </TableHeader>
   391→            <TableBody>
   392→              {MATRIX_VIEWS.map((v) => (
   393→                <TableRow key={v.key}>
   394→                  <TableCell className="pl-1 font-medium text-xs">{v.label}</TableCell>
   395→                  {TENANT_ROLE_LIST.map((r) => (
   396→                    <TableCell key={r} className="text-center">
   397→                      <Checkbox
   398→                        checked={(matrix[v.key] ?? []).includes(r)}
   399→                        onCheckedChange={() => toggle(v.key, r)}
   400→                        aria-label={`${v.label} para ${TENANT_ROLE_LABELS[r]}`}
   401→                      />
   402→                    </TableCell>
   403→                  ))}
   404→                </TableRow>
   405→              ))}
   406→            </TableBody>
   407→          </Table>
   408→        </div>
   409→        <DialogFooter className="gap-2">
   410→          <Button
   411→            variant="outline"
   412→            onClick={() => {
   413→              const defaults: Record<string, string[]> = {}
   414→              for (const v of MATRIX_VIEWS) defaults[v.key] = [...DEFAULT_VIEW_ROLES[v.key]]
   415→              setMatrix(defaults)
   416→              save.mutate({}) // limpa overrides → volta a herdar o padrão
   417→            }}
   418→            disabled={save.isPending}
   419→          >
   420→            Restaurar padrão
   421→          </Button>
   422→          <Button onClick={() => save.mutate(matrix)} disabled={save.isPending}>
   423→            {save.isPending ? <Loader2 className="h-4 w-4 animate-spin" /> : null}
   424→            Salvar permissões
   425→          </Button>
   426→        </DialogFooter>
   427→      </DialogContent>
   428→    </Dialog>
   429→  )
   430→}
