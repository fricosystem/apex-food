    66→function toggleChip(notes: string, chip: string): string {
    67→  const parts = notes.split(',').map((s) => s.trim()).filter(Boolean)
    68→  const idx = parts.findIndex((p) => p.toLowerCase() === chip.toLowerCase())
    69→  if (idx >= 0) parts.splice(idx, 1)
    70→  else parts.push(chip)
    71→  return parts.join(', ')
    72→}

    86→ * Contador animado do total acumulado — conta de R$ 0 até o valor atual na entrada

    90→function useAnimatedMoney(target: number, duration = 900): number {
    91→  const [value, setValue] = useState(0)
    92→  const fromRef = useRef(0)
    93→
    94→  useEffect(() => {
    95→    const from = fromRef.current
    96→    // Diferença irrelevante → conclui em um frame, sem cascata de renders
    97→    const dur = Math.abs(target - from) < 0.004 ? 0 : duration
    98→    let raf = 0
    99→    const start = performance.now()
   100→    const step = (now: number) => {
   101→      const t = dur === 0 ? 1 : Math.min(1, (now - start) / dur)
   102→      const eased = 1 - Math.pow(1 - t, 3) // easeOutCubic — desacelera no fim, estilo caixa registradora
   103→      const v = from + (target - from) * eased
   104→      fromRef.current = v
   105→      setValue(v)
   106→      if (t < 1) raf = requestAnimationFrame(step)
   107→      else {
   108→        fromRef.current = target
   109→        setValue(target)
   110→      }
   111→    }
   112→    raf = requestAnimationFrame(step)
   113→    return () => cancelAnimationFrame(raf)
   114→  }, [target, duration])
   115→
   116→  return value
   117→}

   191→export function ClientView({ token, onExit }: { token: string; onExit: () => void }) {
   192→  const [phase, setPhase] = useState<0 | 1 | 2 | 3 | 4>(0)
   193→  const [cart, setCart] = useState<CartLine[]>([])
   194→  const [activeCat, setActiveCat] = useState<string | null>(null)
   195→  const [tick, setTick] = useState(0)
   196→  const [installEvt, setInstallEvt] = useState<InstallPromptEvent | null>(null)
   197→
   198→  // Tela do cliente sempre em tema escuro (igual à autenticação), independentemente do tema do app
   199→  useEffect(() => {
   200→    const root = document.documentElement
   201→    const wasDark = root.classList.contains('dark')
   202→    root.classList.add('dark')
   203→    return () => {
   204→      if (!wasDark) root.classList.remove('dark')
   205→    }
   206→  }, [])
   207→
   208→  // PWA da mesa: manifest por mesa + metas de instalação + service worker — apenas no modo cliente.
   209→  // Em navegadores sem beforeinstallprompt (ex.: iOS), o cliente usa "Adicionar à Tela de Início".
   210→  useEffect(() => {
   211→    const cleanupPwa = setupClientPwa(token)
   212→    if (isStandalone()) return cleanupPwa
   213→    const onPrompt = (e: Event) => {
   214→      e.preventDefault()
   215→      setInstallEvt(e as InstallPromptEvent)
   216→    }
   217→    const onInstalled = () => {
   218→      setInstallEvt(null)
   219→      toast.success('APEX FOOD instalado no seu celular!')
   220→    }
   221→    window.addEventListener('beforeinstallprompt', onPrompt)
   222→    window.addEventListener('appinstalled', onInstalled)
   223→    return () => {
   224→      window.removeEventListener('beforeinstallprompt', onPrompt)
   225→      window.removeEventListener('appinstalled', onInstalled)
   226→      cleanupPwa()
   227→    }
   228→  }, [token])
   229→
   230→  const menu = useQuery<MenuResponse>({ queryKey: ['client-menu', token], queryFn: () => api<MenuResponse>(`/api/client/${token}/menu`) })
   231→  const { data, isLoading, refetch, isError } = useQuery<ClientData>({
   232→    queryKey: ['client', token, tick],
   233→    queryFn: () => api<ClientData>(`/api/client/${token}`),
   234→    refetchInterval: 3000,
   235→    retry: 2,
   236→  })
   237→
   238→  const productsMap = useMemo(() => {
   239→    const map = new Map<string, Product>()
   240→    for (const c of menu.data?.categories ?? []) for (const p of c.products) map.set(p.id, p)
   241→    return map
   242→  }, [menu.data])
   243→
   244→  const orderId = data?.order?.id
   245→
   246→  // Realtime — sala da comanda do cliente
   247→  useEffect(() => {
   248→    const socket = getSharedSocket()
   249→    if (!orderId) return
   250→    socket.emit('join', { clientRoom: `client:${orderId}` })
   251→    const onChange = () => setTick((t) => t + 1)
   252→    socket.on('item:atualizado', onChange)
   253→    socket.on('comanda:paga', onChange)
   254→    socket.on('comanda:confirmada', onChange)
   255→    return () => {
   256→      socket.off('item:atualizado', onChange)
   257→      socket.off('comanda:paga', onChange)
   258→      socket.off('comanda:confirmada', onChange)
   259→    }
   260→  }, [orderId])
   261→
   262→  // Fase efetivamente exibida (deriva do estado real da comanda, sem efeitos)
   263→  const liveOrder = data?.order ?? null
   264→  const displayPhase: 0 | 1 | 2 | 3 | 4 =
   265→    liveOrder?.status === 'AWAITING_PAYMENT'
   266→      ? 4
   267→      : !liveOrder && data?.lastPaid && phase >= 3
   268→        ? 4
   269→        : phase
   270→
   271→  const finish = useMutation({
   272→    mutationFn: () => apiPost(`/api/client/${token}`, { action: 'finish', orderId }),
   273→    onSuccess: () => {
   274→      playSound('success')
   275→      toast.success('Comanda enviada ao caixa!')
   276→      void refetch()
   277→    },
   278→    onError: (e: Error) => toast.error(e.message),
   279→  })
   280→
   281→  if (isLoading) {
   282→    return (
   283→      <div className="min-h-screen flex items-center justify-center text-zinc-300" style={{ background: BG_DARK }}>
   284→        <div className="flex items-center gap-2 text-zinc-400 text-sm">
   285→          <Loader2 className="h-5 w-5 animate-spin" /> Conectando à mesa…
   286→        </div>
   287→      </div>
   288→    )
   289→  }
   290→
   291→  if (isError || !data) {
   292→    return (
   293→      <div className="min-h-screen flex items-center justify-center p-6" style={{ background: BG_DARK }}>
   294→        <Card className="max-w-sm w-full text-center p-8">
   295→          <p className="text-4xl">🚨</p>
   296→          <p className="font-semibold mt-3">Mesa indisponível</p>
   297→          <p className="text-sm text-muted-foreground mt-1">Não foi possível identificar o QR Code. Chame um funcionário.</p>
   298→          <Button className="mt-6" variant="outline" onClick={onExit}>Painel da equipe</Button>
   299→        </Card>
   300→      </div>
   301→    )
   302→  }
   303→
   304→  const cartCount = cart.reduce((a, c) => a + c.quantity, 0)
   305→  const cartSum = cart.reduce((a, c) => a + (productsMap.get(c.productId)?.price ?? 0) * c.quantity, 0)
   306→
   307→  return (
   308→    <div className="apex-client-decor relative flex min-h-screen flex-col overflow-x-hidden text-zinc-100" style={{ background: BG_DARK }}>
   309→      {/* Luzes ambiente — assinatura premium da mesa (sem interferir nos toques) */}
   310→      <div aria-hidden className="pointer-events-none fixed inset-0 overflow-hidden">
   311→        <div className="apex-orb absolute -left-20 -top-24 h-72 w-72 rounded-full bg-[#FF6B1A]/[0.13] blur-3xl" />
   312→        <div className="apex-orb absolute -right-24 top-1/3 h-80 w-80 rounded-full bg-[#FF8A3D]/[0.09] blur-3xl" style={{ animationDelay: '1.6s' }} />
   313→        <div className="apex-orb absolute bottom-0 left-1/4 h-64 w-64 rounded-full bg-emerald-500/[0.05] blur-3xl" style={{ animationDelay: '3.2s' }} />
   314→      </div>
   315→      <main className="relative mx-auto w-full max-w-md flex-1 p-4 pb-32">
   316→        <PhaseRail phase={displayPhase} tableNumber={data.table.number} />
   317→        {phase === 0 && (
   318→          <WelcomePhase
   319→            tableNumber={data.table.number}
   320→            hasOpen={!!data.order}
   321→            onStart={() => (data.order ? setPhase(3) : setPhase(1))}
   322→            installAvailable={!!installEvt}
   323→            onInstall={() => {
   324→              void installEvt?.prompt()
   325→              setInstallEvt(null)
   326→            }}
   327→          />
   328→        )}
   329→        {phase === 1 && (
   330→          <MenuPhase
   331→            productsMap={productsMap}
   332→            menu={menu.data?.categories ?? []}
   333→            activeCat={activeCat}
   334→            setActiveCat={setActiveCat}
   335→            cart={cart}
   336→            onAdd={(line) => {
   337→              setCart((prev) => {
   338→                const existing = prev.find((c) => c.productId === line.productId && c.notes === line.notes)
   339→                if (existing) {
   340→                  return prev.map((c) => (c.uid === existing.uid ? { ...c, quantity: c.quantity + line.quantity } : c))
   341→                }
   342→                return [...prev, line]
   343→              })
   344→            }}
   345→          />
   346→        )}
   347→        {phase === 2 && (
   348→          <ReviewPhase
   349→            cart={cart}
   350→            productsMap={productsMap}
   351→            onCartChange={setCart}
   352→            onBack={() => setPhase(1)}
   353→            onSent={() => {
   354→              setCart([])
   355→              setPhase(3)
   356→              void refetch()
   357→            }}
   358→            tableToken={token}
   359→            hasOpenOrder={!!liveOrder}
   360→            openTotal={liveOrder?.total ?? 0}
   361→          />
   362→        )}
   363→        {displayPhase === 3 && liveOrder && (
   364→          <TrackingPhase
   365→            order={liveOrder}
   366→            onFinish={() => finish.mutate()}
   367→            finishing={finish.isPending}
   368→            onOrderMore={() => setPhase(1)}
   369→          />
   370→        )}
   371→        {displayPhase === 4 && (
   372→          <ClosingPhase data={data} tick={tick} token={token} onNew={() => { void refetch(); setPhase(1) }} onRated={() => void refetch()} />
   373→        )}
   374→      </main>
   375→
   376→      {(phase === 1 || phase === 2) && (
   377→        <div className="fixed bottom-0 inset-x-0 z-20 border-t border-white/10 bg-[#0B0B0F]/85 p-3 backdrop-blur-xl">
   378→          <div className="max-w-md mx-auto flex items-center gap-2">
   379→            {phase === 2 && (
   380→              <Button variant="outline" onClick={() => setPhase(1)} className="h-11 border-white/15 bg-white/[0.04] text-zinc-200 hover:bg-white/[0.08] hover:text-white">
   381→                <ChevronLeft className="h-4 w-4" /> Cardápio
   382→              </Button>
   383→            )}
   384→            {phase === 1 && (
   385→              <Button
   386→                onClick={() => setPhase(2)}
   387→                disabled={cart.length === 0}
   388→                className="apex-gradient flex h-11 flex-1 items-center justify-center font-semibold text-white transition-transform active:scale-[0.98] disabled:opacity-50"
   389→              >
   390→                <ClipboardList className="h-4 w-4" />
   391→                Revisar comanda
   392→                <span className="mx-1">·</span> <span className="tabular-nums">{currency(cartSum)}</span>
   393→                {cartCount > 0 && <Badge className="ml-1 bg-white/20 text-white hover:bg-white/20">{cartCount}</Badge>}
   394→              </Button>
   395→            )}
   396→          </div>
   397→        </div>
   398→      )}
   399→
   400→      <p className="relative pb-4 text-center text-[10px] text-muted-foreground">
   401→        <button className="underline" onClick={onExit}>Painel da equipe</button>
   402→      </p>
   403→    </div>
   404→  )
   405→}

   875→function TrackingPhase({ order, onFinish, finishing, onOrderMore }: { order: ClientOrder; onFinish: () => void; finishing: boolean; onOrderMore: () => void }) {
   876→  const [now, setNow] = useState(Date.now())
   877→
   878→  useEffect(() => {
   879→    const t = setInterval(() => setNow(Date.now()), 1000)
   880→    return () => clearInterval(t)
   881→  }, [])
