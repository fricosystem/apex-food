   114→   114→const clamp01 = (v: number) => Math.min(1, Math.max(0, v))
   115→   115→
   116→   116→/**
   117→   117→ * Fade vinculado à posição de rolagem (sem transição CSS — o estilo é recalculado a cada frame):
   118→   118→ * - rolando para baixo: o que entra por baixo sofre fade in subindo (+44px → 0) e o que
   119→   119→ *   sai por cima sofre fade out (0 → -30px);
   120→   120→ * - rolando para cima (reverse): o mesmo cálculo posicional inverte o efeito — o conteúdo
   121→   121→ *   superior volta com fade in descendo e o inferior some com fade out.
   122→   122→ */
   123→   123→type FadeApi = { register: (el: HTMLDivElement | null) => void }
   124→   124→const FadeCtx = createContext<FadeApi | null>(null)
   125→   125→
   126→   126→function useFadeController(): FadeApi {
   127→   127→  const items = useRef(new Set<HTMLDivElement>())
   128→   128→  const rafRef = useRef(0)
   129→   129→
   130→   130→  const update = useCallback(() => {
   131→   131→    rafRef.current = 0
   132→   132→    const H = window.innerHeight || 1
   133→   133→    const reduced = window.matchMedia('(prefers-reduced-motion: reduce)').matches
   134→   134→    items.current.forEach((el) => {
   135→   135→      if (!el.isConnected) {
   136→   136→        items.current.delete(el)
   137→   137→        return
   138→   138→      }
   139→   139→      if (reduced) {
   140→   140→        el.style.opacity = '1'
   141→   141→        el.style.transform = ''
   142→   142→        return
   143→   143→      }
   144→   144→      const rect = el.getBoundingClientRect()
   145→   145→      // Desconta o transform aplicado no frame anterior para medir a posição de layout
   146→   146→      const prevY = Number(el.dataset.fadeY ?? 0)
   147→   147→      const top = rect.top - prevY
   148→   148→      const bottom = rect.bottom - prevY
   149→   149→      // Entrada: topo percorrendo os 22% inferiores da viewport
   150→   150→      const tIn = clamp01((H - top) / (H * 0.22))
   151→   151→      // Saída: base adentrando os 28% superiores da viewport
   152→   152→      const tOut = clamp01(bottom / (H * 0.28))
   153→   153→      // Blocos ancorados na primeira dobra nascem visíveis e sofrem apenas o fade de saída
   154→   154→      const anchored = el.dataset.fadeAnchored === '1'
   155→   155→      const opacity = anchored ? tOut : Math.min(tIn, tOut)
   156→   156→      const y = (anchored ? 0 : (1 - tIn) * 44) - (1 - tOut) * 30
   157→   157→      el.dataset.fadeY = y.toFixed(1)
   158→   158→      el.style.opacity = opacity.toFixed(3)
   159→   159→      el.style.transform = `translate3d(0, ${y.toFixed(1)}px, 0)`
   160→   160→    })
   161→   161→  }, [])
   162→   162→
   163→   163→  const schedule = useCallback(() => {
   164→   164→    if (!rafRef.current) rafRef.current = requestAnimationFrame(update)
   165→   165→  }, [update])
   166→   166→
   167→   167→  const register = useCallback((el: HTMLDivElement | null) => {
   168→   168→    if (el) items.current.add(el)
   169→   169→  }, [])
   170→   170→
   171→   171→  useEffect(() => {
   172→   172→    schedule()
   173→   173→    window.addEventListener('scroll', schedule, { passive: true })
   174→   174→    window.addEventListener('resize', schedule)
   175→   175→    // Recalcula após fontes/imagens estabilizarem o layout
   176→   176→    const t1 = window.setTimeout(schedule, 350)
   177→   177→    const t2 = window.setTimeout(schedule, 900)
   178→   178→    document.fonts?.ready.then(schedule).catch(() => {})
   179→   179→    return () => {
   180→   180→      window.removeEventListener('scroll', schedule)
   181→   181→      window.removeEventListener('resize', schedule)
   182→   182→      window.clearTimeout(t1)
   183→   183→      window.clearTimeout(t2)
   184→   184→      if (rafRef.current) cancelAnimationFrame(rafRef.current)
   185→   185→    }
   186→   186→  }, [schedule])
   187→   187→
   188→   188→  return { register }
   189→   189→}

   191→   191→/** Bloco que participa do fade de rolagem; startVisible mantém o bloco visível antes da hidratação e anchored desativa o fade de entrada (só sai por cima) */
   192→   192→function ScrollFade({ children, className, startVisible, anchored }: {
   193→   193→  children: React.ReactNode
   194→   194→  className?: string
   195→   195→  startVisible?: boolean
   196→   196→  anchored?: boolean
   197→   197→}) {
   198→   198→  const fade = useContext(FadeCtx)
   199→   199→  return (
   200→   200→    <div
   201→   201→      ref={fade?.register}
   202→   202→      className={cn('relative', className)}
   203→   203→      data-fade-anchored={anchored ? '1' : undefined}
   204→   204→      data-fade-y={startVisible ? undefined : '44'}
   205→   205→      style={{
   206→   206→        opacity: startVisible ? undefined : 0,
   207→   207→        transform: startVisible ? undefined : 'translate3d(0, 44px, 0)',
   208→   208→        willChange: 'opacity, transform',
   209→   209→      }}
   210→   210→    >
   211→   211→      {children}
   212→   212→    </div>
   213→   213→  )
   214→   214→}
