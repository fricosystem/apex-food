    64→    64→function toggleChip(notes: string, chip: string): string {
    65→    65→  const parts = notes.split(',').map((s) => s.trim()).filter(Boolean)
    66→    66→  const idx = parts.findIndex((p) => p.toLowerCase() === chip.toLowerCase())
    67→    67→  if (idx >= 0) parts.splice(idx, 1)
    68→    68→  else parts.push(chip)
    69→    69→  return parts.join(', ')
    70→    70→}

   236→   236→            productsMap={productsMap}
   237→   237→            onCartChange={setCart}
   238→   238→            onBack={() => setPhase(1)}
   239→   239→            onSent={() => {

   241→   241→              setPhase(3)

   277→   277→      )}
   278→   278→
   279→   279→      <p className="pb-4 text-center text-[10px] text-muted-foreground">
   280→   280→        <button className="underline" onClick={onExit}>Painel da equipe</button>
   281→   281→      </p>
   282→   282→    </div>
   283→   283→  )

   367→   367→function MenuPhase({
   368→   368→  menu, activeCat, setActiveCat, cart, onAdd, productsMap,
   369→   369→}: {
   370→   370→  menu: Array<Category & { products: Product[] }>

   378→   378→

   380→   380→    return <div className="text-center text-muted-foreground text-sm py-16">Cardápio em atualização…</div>
   381→   381→  }
   382→   382→
   383→   383→  const countInCart = (id: string) => cart.find((c) => c.productId === id)?.quantity ?? 0
   384→   384→
   385→   385→  return (
   386→   386→    <div className="apex-enter">

   391→   391→            onClick={() => setActiveCat(c.id)}

   393→   393→              'shrink-0 rounded-full border px-3.5 py-2 text-xs font-medium transition-all flex items-center gap-1.5',

   397→   397→            )}

   401→   401→        ))}

   403→   403→
   404→   404→      <div className="mt-4 space-y-3">
   405→   405→        {current?.products.map((p) => (
   406→   406→          <ProductCard key={p.id} product={p} inCart={countInCart(p.id)} onAdd={onAdd} />
