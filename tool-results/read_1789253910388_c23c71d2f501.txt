    55→/** Aviso de exemplo usado no botão de teste de cada tipo de ação */
    56→function sampleFor(kind: NotifKind): { title: string; body: string } {
    57→  switch (kind) {
    58→    case 'comanda-nova':
    59→      return { title: 'Nova comanda — Mesa 12', body: 'Teste de notificação: cliente abriu a comanda pelo QR Code.' }
    60→    case 'comanda-confirmada':
    61→      return { title: 'Comanda confirmada na cozinha', body: 'Teste de notificação: 3 itens entraram na fila de preparo.' }
    62→    case 'prato-pronto':
    63→      return { title: 'Prato pronto para servir', body: 'Teste de notificação: Pizza Margherita — Mesa 07.' }
    64→    case 'encaminhada':
    65→      return { title: 'Mesa 07 encaminhada ao caixa', body: 'Teste de notificação: comanda aguardando pagamento.' }
    66→    case 'pagamento':
    67→      return { title: 'Pagamento confirmado — Mesa 07', body: 'Teste de notificação: comanda #A31 finalizada via PIX.' }
    68→    case 'alerta':
    69→      return { title: 'Alerta de operação', body: 'Teste de notificação: comanda da Mesa 03 acima de 45 min sem itens servidos.' }
    70→  }
    71→}

    73→export function SettingsView({ user }: { user: SessionUser }) {
    74→  const qc = useQueryClient()
    75→  const { theme, setTheme } = useTheme()
    76→  const { setActiveView } = useAppStore()
    77→
    78→  const isCompact = user.role === 'WAITER' || user.role === 'KITCHEN' || user.role === 'CASHIER'
    79→  const isAdminish = user.role === 'ADMIN' || user.role === 'MANAGER'
    80→
    81→  const [formEdits, setFormEdits] = useState<{ name?: string; type?: string; logo?: string }>({})
    82→

    84→  // então inicializadores lazy podem ler localStorage sem mismatch de hidratação
    85→  const [permState, setPermState] = useState<NotificationPermission | 'unsupported'>(() => notificationPermission())
    86→  const [notifMaster, setNotifMaster] = useState(() => isSystemNotifEnabled())

    93→  })

   105→      ?.query({ name: 'notifications' as PermissionName })
   106→      .then((s) => {
   107→        st = s
   108→        s.onchange = onChange

   112→      if (st) st.onchange = null
   113→    }

   115→
   116→  const { data, isLoading } = useQuery<{ settings: Record<string, string> }>({
   117→    queryKey: ['settings'],
   118→    queryFn: () => api('/api/settings'),
   119→  })

   121→    queryKey: ['users'],
   122→    queryFn: () => api('/api/users'),
   123→    enabled: isAdminish,
   124→  })
   125→
   126→  const s = data?.settings ?? {}
   127→  const form = {

   129→    establishmentType: formEdits.type ?? s.establishmentType ?? 'RESTAURANTE',
   130→    establishmentLogo: formEdits.logo ?? s.establishmentLogo ?? '🍴',
   131→  }
   132→
   133→  const saveSettings = useMutation({
   134→    mutationFn: (payload: Record<string, string>) => apiPatch('/api/settings', payload),
   135→    onSuccess: () => {
   136→      void qc.invalidateQueries({ queryKey: ['settings'] })

   139→    onError: (e: Error) => toast.error(e.message),
   140→  })
   141→

   143→    setFormEdits((prev) => ({ ...prev, ...patch }))

   176→        toast.warning('Notificação do sistema desligada para este tipo', {
   177→          description: 'Ative o interruptor do tipo de ação para receber o popup.',

   222→        </div>
   223→        <div className="flex-1 min-w-0">
   224→          <div className="flex items-center gap-2 flex-wrap">
   225→            <h2 className="text-lg font-bold tracking-tight truncate">{form.establishmentName}</h2>
   226→            <Badge className="apex-gradient text-white border-0 text-[10px]">Sistema ativo</Badge>

   228→          <p className="text-xs text-muted-foreground mt-1">
   229→            {EST_TYPES.find((t) => t.key === form.establishmentType)?.label ?? 'Restaurante'} · 10 módulos · 5 perfis de acesso · operação em tempo real

   237→      </div>
   238→
   239→      {/* Notificações do dispositivo — desktop, tablet e mobile PWA */}
   240→      <Card className="lg:col-span-2 border-primary/25">
   241→        <CardContent className="p-5 space-y-4">
   242→          <div className="flex items-center justify-between gap-2 flex-wrap">
   243→            <div className="flex items-center gap-2">
   244→              <BellRing className="text-primary" style={{ width: 18, height: 18 }} />
   245→              <p className="font-semibold">Notificações do dispositivo</p>
   246→            </div>
   247→            <div className="flex items-center gap-2">
   248→              <Badge variant="outline" className={cn('text-[10px]', permBadge.cls)}>
   249→                {permBadge.label}
   250→              </Badge>
   251→              <Badge variant="outline" className="text-[10px] text-muted-foreground border-border gap-1">
   252→                <MonitorSmartphone className="h-3 w-3" /> Desktop · Tablet · PWA
   253→              </Badge>
   254→            </div>
   255→          </div>
   256→
   257→          {/* Estado da permissão */}
   258→          {permState === 'unsupported' ? (
   259→            <div className="flex items-start gap-2.5 rounded-lg border border-amber-500/30 bg-amber-500/5 p-3">
   260→              <AlertTriangle className="h-4 w-4 text-amber-500 shrink-0 mt-0.5" />
   261→              <p className="text-xs text-muted-foreground">
   262→                Este navegador não suporta notificações do sistema. No iPhone/iPad, instale o app pela tela do
   263→                cliente (Compartilhar → Adicionar à Tela de Início) e abra pelo ícone para receber os avisos.
   264→              </p>
   265→            </div>
   266→          ) : permState === 'denied' ? (
   267→            <div className="flex items-start gap-2.5 rounded-lg border border-red-500/30 bg-red-500/5 p-3">

   269→              <p className="text-xs text-muted-foreground">

   271→                notificações para voltar a receber os avisos na barra do sistema.
   272→              </p>
   273→            </div>
   274→          ) : permState === 'default' ? (
   275→            <div className="flex flex-col sm:flex-row sm:items-center gap-3 rounded-lg border border-primary/30 bg-primary/5 p-3">
   276→              <div className="flex items-start gap-2.5 flex-1">

   289→              <Check className="h-4 w-4 text-emerald-500 shrink-0" />

   293→              <Button size="sm" variant="outline" className="shrink-0 gap-1.5" onClick={() => testKind('prato-pronto')}>
   294→                <Play className="h-3.5 w-3.5" /> Testar agora
   295→              </Button>
   296→            </div>

   299→          {/* Master das notificações do sistema */}
   300→          <div className="flex items-center justify-between rounded-lg border p-3">
   301→            <div className="flex items-center gap-2.5">

   305→                <p className="text-[11px] text-muted-foreground">
   306→                  Popup na tela e na central de notificações (desktop, tablet e PWA no celular)

   312→              onCheckedChange={(v) => {
   313→                setNotifMaster(v)
   314→                setSystemNotifEnabled(v)
   315→                playSound('click')
   316→                toast.success(v ? 'Notificações do sistema ativadas' : 'Notificações do sistema desativadas')
   317→              }}
   318→              aria-label="Notificações do sistema"
   319→            />
   320→          </div>
   321→
   322→          <Separator />
   323→
   324→          {/* Tipos de ação: som + popup por tipo */}
   325→          <div>
   326→            <p className="text-xs font-semibold text-muted-foreground mb-2">
   327→              Tipos de ação — som personalizado e popup para cada um
   328→            </p>
   329→            <div className="rounded-lg border divide-y">
   330→              {kindList.map((kind) => {
   331→                const meta = NOTIF_KINDS[kind]
   332→                const on = kindNotif[kind] !== false

   388→              const active = theme === t.key
   389→              return (
   390→                <button

   404→          <div className="flex items-center justify-between rounded-lg border p-3">
   405→            <div className="flex items-center gap-2.5">

   410→              </div>
   411→            </div>
   412→            <Badge variant="outline" className="text-[10px] text-muted-foreground border-border">#FF6B1A</Badge>

   417→      {/* Sons e volume */}
   418→      <Card>
   419→        <CardContent className="p-5 space-y-4">
   420→          <div className="flex items-center gap-2">
   421→            {sound ? <Volume2 className="text-primary" style={{ width: 18, height: 18 }} /> : <VolumeX className="text-muted-foreground" style={{ width: 18, height: 18 }} />}

   428→                <p className="text-sm font-medium">Sons de alerta</p>

   431→            </div>

   438→              }}
   439→              aria-label="Sons"
   440→            />
   441→          </div>
   442→          <div className="rounded-lg border p-3 space-y-2.5">
