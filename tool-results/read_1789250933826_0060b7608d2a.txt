    52→    52→/** Número com animação count-up (ease-out cúbico), respeitando prefers-reduced-motion */
    53→    53→function AnimatedNumber({ value, format }: { value: number; format?: (v: number) => string }) {
    54→    54→  const [display, setDisplay] = useState(value)
    55→    55→  const prevRef = useRef(value)
    56→    56→
    57→    57→  useEffect(() => {
    58→    58→    const from = prevRef.current
    59→    59→    const to = value
    60→    60→    prevRef.current = value
    61→    61→    if (from === to) return
    62→    62→    if (typeof window !== 'undefined' && window.matchMedia?.('(prefers-reduced-motion: reduce)').matches) {
    63→    63→      const id = requestAnimationFrame(() => setDisplay(to))
    64→    64→      return () => cancelAnimationFrame(id)
    65→    65→    }
    66→    66→    const duration = 900
    67→    67→    let raf = 0
    68→    68→    const t0 = performance.now()
    69→    69→    const tick = (t: number) => {
    70→    70→      const p = Math.min(1, (t - t0) / duration)
    71→    71→      const eased = 1 - Math.pow(1 - p, 3)
    72→    72→      setDisplay(from + (to - from) * eased)
    73→    73→      if (p < 1) raf = requestAnimationFrame(tick)
    74→    74→    }
    75→    75→    raf = requestAnimationFrame(tick)
    76→    76→    return () => cancelAnimationFrame(raf)
    77→    77→  }, [value])
    78→    78→
    79→    79→  return <>{format ? format(display) : Math.round(display).toLocaleString('pt-BR')}</>
    80→    80→}

    82→    82→/** Badge de variação vs. período anterior (verde ↑ / vermelho ↓) */
    83→    83→function DeltaBadge({ current, previous }: { current: number; previous: number }) {
    84→    84→  if (!previous || previous <= 0) return null
    85→    85→  const pct = ((current - previous) / previous) * 100
    86→    86→  const up = pct >= 0
    87→    87→  return (
    88→    88→    <span
    89→    89→      className={cn(
    90→    90→        'inline-flex items-center gap-1 rounded-full px-2 py-0.5 text-[10px] font-semibold border',
    91→    91→        up
    92→    92→          ? 'bg-emerald-500/10 text-emerald-600 dark:text-emerald-400 border-emerald-500/30'
    93→    93→          : 'bg-red-500/10 text-red-600 dark:text-red-400 border-red-500/30',
    94→    94→      )}
    95→    95→    >
    96→    96→      {up ? <TrendingUp className="h-3 w-3" /> : <TrendingDown className="h-3 w-3" />}
    97→    97→      {up ? '+' : ''}{pct.toFixed(1)}%
    98→    98→    </span>
    99→    99→  )
   100→   100→}
