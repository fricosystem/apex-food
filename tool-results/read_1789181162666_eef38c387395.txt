    51→export function ClientView({ token, onExit }: { token: string; onExit: () => void }) {
    52→  const [phase, setPhase] = useState<0 | 1 | 2 | 3 | 4>(0)
    53→  const [cart, setCart] = useState<CartLine[]>([])
    54→  const [activeCat, setActiveCat] = useState<string | null>(null)
    55→  const [tick, setTick] = useState(0)
    56→
    57→  const menu = useQuery<MenuResponse>({ queryKey: ['categories'], queryFn: () => api<MenuResponse>('/api/categories') })
    58→  const { data, isLoading, refetch, isError } = useQuery<ClientData>({
    59→    queryKey: ['client', token, tick],
    60→    queryFn: () => api<ClientData>(`/api/client/${token}`),
    61→    refetchInterval: 3000,
    62→    retry: 2,
    63→  })
    64→
    65→  const productsMap = useMemo(() => {
    66→    const map = new Map<string, Product>()
    67→    for (const c of menu.data?.categories ?? []) for (const p of c.products) map.set(p.id, p)
    68→    return map
    69→  }, [menu.data])
    70→
    71→  const orderId = data?.order?.id
    72→
    73→  // Realtime — sala da comanda do cliente
    74→  useEffect(() => {
    75→    const socket = getSharedSocket()
    76→    if (!orderId) return
    77→    socket.emit('join', { clientRoom: `client:${orderId}` })
    78→    const onChange = () => setTick((t) => t + 1)
    79→    socket.on('item:atualizado', onChange)
    80→    socket.on('comanda:paga', onChange)
    81→    socket.on('comanda:confirmada', onChange)
    82→    return () => {
    83→      socket.off('item:atualizado', onChange)
    84→      socket.off('comanda:paga', onChange)
    85→      socket.off('comanda:confirmada', onChange)
    86→    }
    87→  }, [orderId])
    88→
    89→  // Fase efetivamente exibida (deriva do estado real da comanda, sem efeitos)
    90→  const liveOrder = data?.order ?? null
    91→  const displayPhase: 0 | 1 | 2 | 3 | 4 =
    92→    liveOrder?.status === 'AWAITING_PAYMENT'
    93→      ? 4
    94→      : !liveOrder && data?.lastPaid && phase >= 3
    95→        ? 4
    96→        : phase
    97→
    98→  const finish = useMutation({
    99→    mutationFn: () => apiPost(`/api/client/${token}`, { action: 'finish', orderId }),
   100→    onSuccess: () => {
   101→      playSound('success')
   102→      toast.success('Comanda enviada ao caixa!')
   103→      void refetch()
   104→    },
   105→    onError: (e: Error) => toast.error(e.message),
   106→  })
   107→
   108→  if (isLoading) {
   109→    return (
   110→      <div className="min-h-screen bg-background flex items-center justify-center">
   111→        <div className="flex items-center gap-2 text-muted-foreground text-sm">
   112→          <Loader2 className="h-5 w-5 animate-spin" /> Conectando à mesa…
   113→        </div>
   114→      </div>
   115→    )
   116→  }
   117→
   118→  if (isError || !data) {
   119→    return (
   120→      <div className="min-h-screen bg-background flex items-center justify-center p-6">
   121→        <Card className="max-w-sm w-full text-center p-8">
   122→          <p className="text-4xl">🚨</p>
   123→          <p className="font-semibold mt-3">Mesa indisponível</p>
   124→          <p className="text-sm text-muted-foreground mt-1">Não foi possível identificar o QR Code. Chame um funcionário.</p>
   125→          <Button className="mt-6" variant="outline" onClick={onExit}>Painel da equipe</Button>
   126→        </Card>
   127→      </div>
   128→    )
   129→  }
   130→
   131→  const cartCount = cart.reduce((a, c) => a + c.quantity, 0)
   132→  const cartSum = cart.reduce((a, c) => a + (productsMap.get(c.productId)?.price ?? 0) * c.quantity, 0)
   133→
   134→  return (
   135→    <div className="min-h-screen bg-background flex flex-col">
   136→      <header className="sticky top-0 z-20 apex-gradient text-white shadow-lg">
   137→        <div className="max-w-md mx-auto px-4 h-14 flex items-center gap-3">
   138→          <div className="h-8 w-8 rounded-lg bg-white/15 ring-1 ring-white/25 flex items-center justify-center">
   139→            <img src="/apex-logo.png" alt="Logo APEX FOOD" className="h-5 w-5 object-contain" />
   140→          </div>
   141→          <div className="flex-1 min-w-0">
   142→            <p className="text-sm font-bold leading-none">Mesa {String(data.table.number).padStart(2, '0')}</p>
   143→            <p className="text-[11px] text-white/75 mt-0.5">APEX FOOD</p>
   144→          </div>
   145→          {phase === 1 && cartCount > 0 && (
   146→            <Button size="sm" variant="ghost" className="text-white hover:bg-white/15 h-8" onClick={() => setPhase(2)}>
   147→              <ShoppingBag className="h-4 w-4 mr-1" /> {cartCount}
   148→            </Button>
   149→          )}
   150→        </div>
   151→        <div className="max-w-md mx-auto px-4 pb-2.5">
   152→          <div className="flex gap-1">
   153→            {STEPS.map((_, i) => (
   154→              <div key={i} className={cn('h-1.5 flex-1 rounded-full transition-colors duration-300', i <= displayPhase ? 'bg-white' : 'bg-white/25')} />
   155→            ))}
   156→          </div>
   157→          <div className="flex justify-between mt-1.5">
   158→            {STEPS.map((s, i) => (
   159→              <span key={s} className={cn('text-[9px] leading-none', i === displayPhase ? 'text-white font-semibold' : 'text-white/55')}>
   160→                {s}
   161→              </span>
   162→            ))}
   163→          </div>
   164→        </div>
   165→      </header>
   166→
   167→      <main className="flex-1 max-w-md w-full mx-auto p-4 pb-32">
   168→        {phase === 0 && (
   169→          <WelcomePhase tableNumber={data.table.number} hasOpen={!!data.order} onStart={() => (data.order ? setPhase(3) : setPhase(1))} />
   170→        )}
   171→        {phase === 1 && (
   172→          <MenuPhase
   173→            productsMap={productsMap}
   174→            menu={menu.data?.categories ?? []}
   175→            activeCat={activeCat}
   176→            setActiveCat={setActiveCat}
   177→            cart={cart}
   178→            onAdd={(line) => {
   179→              setCart((prev) => {
   180→                const existing = prev.find((c) => c.productId === line.productId)
   181→                if (existing) {
   182→                  return prev.map((c) => (c.productId === line.productId ? { ...line, quantity: c.quantity + line.quantity } : c))
   183→                }
   184→                return [...prev, line]
   185→              })
   186→            }}
   187→          />
   188→        )}
   189→        {phase === 2 && (
   190→          <ReviewPhase
   191→            cart={cart}
   192→            productsMap={productsMap}
   193→            onCartChange={setCart}
   194→            onBack={() => setPhase(1)}
   195→            onSent={() => {
   196→              setCart([])
   197→              setPhase(3)
   198→              void refetch()
   199→            }}
   200→            tableToken={token}
   201→          />
   202→        )}
   203→        {displayPhase === 3 && liveOrder && (
   204→          <TrackingPhase order={liveOrder} onFinish={() => finish.mutate()} finishing={finish.isPending} />
   205→        )}
   206→        {displayPhase === 4 && (
   207→          <ClosingPhase data={data} tick={tick} onNew={() => { void refetch(); setPhase(1) }} />
   208→        )}
   209→      </main>
   210→
   211→      {(phase === 1 || phase === 2) && (
   212→        <div className="fixed bottom-0 inset-x-0 border-t bg-background/92 backdrop-blur p-3 z-20">
   213→          <div className="max-w-md mx-auto flex items-center gap-2">
   214→            {phase === 2 && (
   215→              <Button variant="outline" onClick={() => setPhase(1)} className="h-11">
   216→                <ChevronLeft className="h-4 w-4" /> Cardápio
   217→              </Button>
   218→            )}
   219→            {phase === 1 && (
   220→              <Button
   221→                onClick={() => setPhase(2)}
   222→                disabled={cart.length === 0}
   223→                className="flex-1 h-11 apex-gradient text-white font-semibold"
   224→              >
   225→                <ClipboardList className="h-4 w-4" />
   226→                Revisar comanda
   227→                <span className="mx-1">·</span> {currency(cartSum)}
   228→                {cartCount > 0 && <Badge className="ml-1 bg-white/20 text-white hover:bg-white/20">{cartCount}</Badge>}
   229→              </Button>
   230→            )}
   231→          </div>
   232→        </div>
   233→      )}
   234→
   235→      <p className="pb-4 text-center text-[10px] text-muted-foreground">
   236→        <button className="underline" onClick={onExit}>Painel da equipe</button>
   237→      </p>
   238→    </div>
   239→  )
   240→}

   274→/* ==================== FASE 2 — Cardápio ==================== */
   275→function MenuPhase({
   276→  menu, activeCat, setActiveCat, cart, onAdd, productsMap,
   277→}: {
   278→  menu: Array<Category & { products: Product[] }>
   279→  activeCat: string | null
   280→  setActiveCat: (c: string | null) => void
   281→  cart: CartLine[]
   282→  onAdd: (line: CartLine) => void
   283→  productsMap: Map<string, Product>
   284→}) {
   285→  const current = useMemo(() => menu.find((c) => c.id === activeCat) ?? menu[0], [menu, activeCat])
   286→
   287→  if (menu.length === 0) {
   288→    return <div className="text-center text-muted-foreground text-sm py-16">Cardápio em atualização…</div>
   289→  }
   290→
   291→  const countInCart = (id: string) => cart.find((c) => c.productId === id)?.quantity ?? 0
   292→
   293→  return (
   294→    <div className="apex-enter">
   295→      <div className="flex gap-1.5 overflow-x-auto pb-2 -mx-1 px-1">
   296→        {menu.map((c) => (
   297→          <button
   298→            key={c.id}
   299→            onClick={() => setActiveCat(c.id)}
   300→            className={cn(
   301→              'shrink-0 rounded-full border px-3.5 py-2 text-xs font-medium transition-all flex items-center gap-1.5',
   302→              current?.id === c.id
   303→                ? 'border-primary/50 bg-primary/10 text-primary'
   304→                : 'bg-card text-muted-foreground hover:text-foreground'
   305→            )}
   306→          >
   307→            <span>{c.icon}</span> {c.name}
   308→          </button>
   309→        ))}
   310→      </div>
   311→
   312→      <div className="mt-4 space-y-3">
   313→        {current?.products.map((p) => (
   314→          <ProductCard key={p.id} product={p} inCart={countInCart(p.id)} onAdd={onAdd} />
   315→        ))}
   316→        {current?.products.length === 0 && (
   317→          <p className="text-center text-sm text-muted-foreground py-10">Sem itens disponíveis nesta categoria.</p>
   318→        )}
   319→      </div>
   320→
   321→      <div className="text-center text-[10px] text-muted-foreground mt-6">
   322→        {productsMap.size} itens no cardápio · preços em reais
   323→      </div>
   324→    </div>
   325→  )
   326→}

   328→function ProductCard({ product, inCart, onAdd }: { product: Product; inCart: number; onAdd: (line: CartLine) => void }) {
   329→  const [open, setOpen] = useState(false)
   330→  const [qty, setQty] = useState(1)
   331→  const [notes, setNotes] = useState('')
   332→
   333→  const confirmAdd = () => {
   334→    playSound('click')
   335→    onAdd({ productId: product.id, quantity: qty, notes: notes.trim() })
   336→    toast.success(`${product.name} adicionado à comanda`)
   337→    setOpen(false)
   338→  }
   339→
   340→  return (
   341→    <Card className="overflow-hidden transition-shadow hover:shadow-md">
   342→      <CardContent className="p-0 flex">
   343→        {product.image ? (
   344→          <img src={product.image} alt={product.name} className="h-24 w-24 object-cover shrink-0" />
   345→        ) : (
   346→          <div className="h-24 w-24 shrink-0 apex-gradient-soft flex items-center justify-center border-r">
   347→            <span className="text-4xl" aria-hidden>{product.emoji}</span>
   348→          </div>
   349→        )}
   350→        <div className="flex-1 min-w-0 p-3">
   351→          <div className="flex items-start justify-between gap-2">
   352→            <p className="font-semibold text-sm leading-tight">{product.name}</p>
   353→            <Badge variant="outline" className="shrink-0 text-[10px] gap-1">
   354→              <Clock className="h-3 w-3" /> {product.prepTime}min
   355→            </Badge>
   356→          </div>
   357→          <p className="text-xs text-muted-foreground mt-1 line-clamp-2 leading-snug">{product.description}</p>
   358→          <div className="flex items-center justify-between mt-2.5">
   359→            <p className="font-bold text-primary">{currency(product.price)}</p>
   360→            {inCart > 0 ? (
   361→              <Badge className="apex-gradient text-white border-0 gap-1">
   362→                <CheckCircle2 className="h-3 w-3" /> {inCart} na comanda
   363→              </Badge>
   364→            ) : (
   365→              <Button size="sm" className="h-8 apex-gradient text-white" onClick={() => { setQty(1); setNotes(''); setOpen(true) }}>
   366→                <Plus className="h-3.5 w-3.5" /> Adicionar
   367→              </Button>
   368→            )}
   369→          </div>
   370→        </div>
   371→      </CardContent>
   372→
   373→      <Dialog open={open} onOpenChange={setOpen}>
   374→        <DialogContent className="max-w-[340px]">
   375→          <DialogHeader>
   376→            <DialogTitle className="flex items-center gap-2 text-base">
   377→              <span className="text-xl">{product.emoji}</span> {product.name}
   378→            </DialogTitle>
   379→          </DialogHeader>
   380→          <p className="text-xs text-muted-foreground">{product.description}</p>
   381→          <div className="flex items-center gap-3">
   382→            <span className="text-sm">Qtd.</span>
   383→            <div className="flex items-center gap-2">
   384→              <Button size="icon" variant="outline" className="h-8 w-8" onClick={() => setQty((q) => Math.max(1, q - 1))} aria-label="Menos">
   385→                <Minus className="h-3.5 w-3.5" />
   386→              </Button>
   387→              <span className="w-6 text-center font-bold">{qty}</span>
   388→              <Button size="icon" variant="outline" className="h-8 w-8" onClick={() => setQty((q) => Math.min(20, q + 1))} aria-label="Mais">
   389→                <Plus className="h-3.5 w-3.5" />
   390→              </Button>
   391→            </div>
   392→            <p className="ml-auto font-bold text-primary">{currency(product.price * qty)}</p>
   393→          </div>
   394→          <Textarea
   395→            placeholder="Observações (ex: sem cebola, ponto da carne…)"
   396→            value={notes}
   397→            onChange={(e) => setNotes(e.target.value)}
   398→            className="min-h-[70px] text-sm"
   399→          />
   400→          <DialogFooter>
   401→            <Button className="w-full apex-gradient text-white font-semibold" onClick={confirmAdd}>
   402→              <Plus className="h-4 w-4" /> Adicionar {qty} à comanda
   403→            </Button>
   404→          </DialogFooter>
   405→        </DialogContent>
   406→      </Dialog>
   407→    </Card>
   408→  )
   409→}

   411→/* ==================== FASE 3 — Revisão ==================== */
   412→function ReviewPhase({
   413→  cart, productsMap, onCartChange, onBack, onSent, tableToken,
   414→}: {
   415→  cart: CartLine[]
   416→  productsMap: Map<string, Product>
   417→  onCartChange: (c: CartLine[]) => void
   418→  onBack: () => void
   419→  onSent: () => void
   420→  tableToken: string
   421→}) {
   422→  const [editing, setEditing] = useState<CartLine | null>(null)
   423→  const [editQty, setEditQty] = useState(1)
   424→  const [editNotes, setEditNotes] = useState('')
   425→
   426→  const subtotal = cart.reduce((a, c) => a + (productsMap.get(c.productId)?.price ?? 0) * c.quantity, 0)
   427→  const estMinutes = Math.max(...cart.map((c) => productsMap.get(c.productId)?.prepTime ?? 15), 0)
   428→  const itemCount = cart.reduce((a, c) => a + c.quantity, 0)
   429→
   430→  const send = useMutation({
   431→    mutationFn: () =>
   432→      apiPost<{ order: ClientOrder }>('/api/orders', {
   433→        tableToken,
   434→        items: cart.map((c) => ({ productId: c.productId, quantity: c.quantity, notes: c.notes })),
   435→      }),
   436→    onSuccess: () => {
   437→      playSound('success')
   438→      toast.success('Comanda enviada ao garçom!')
   439→      onCartChange([])
   440→      onSent()
   441→    },
   442→    onError: (e: Error) => toast.error(e.message),
   443→  })
   444→
   445→  const changeQty = (productId: string, delta: number) => {
   446→    playSound('click')
   447→    onCartChange(
   448→      cart
   449→        .map((c) => (c.productId === productId ? { ...c, quantity: c.quantity + delta } : c))
   450→        .filter((c) => c.quantity > 0)
   451→    )
   452→  }
   453→
   454→  if (cart.length === 0) {
   455→    return (
   456→      <div className="text-center py-16 apex-enter">
   457→        <ClipboardList className="h-12 w-12 mx-auto text-muted-foreground/40" />
   458→        <p className="font-semibold mt-4">Comanda vazia</p>
   459→        <p className="text-sm text-muted-foreground mt-1">Adicione itens pelo cardápio.</p>
   460→        <Button onClick={onBack} className="mt-6 apex-gradient text-white">Ir ao cardápio</Button>
   461→      </div>
   462→    )
   463→  }
   464→
   465→  return (
   466→    <div className="apex-enter space-y-3">
   467→      <div className="flex items-center justify-between">
   468→        <h2 className="font-bold tracking-tight">Revise sua comanda</h2>
   469→        <Badge variant="outline">{itemCount} item(ns)</Badge>
   470→      </div>
   471→
   472→      {cart.map((line) => {
   473→        const p = productsMap.get(line.productId)
   474→        if (!p) return null
   475→        return (
   476→          <Card key={line.productId}>
   477→            <CardContent className="p-3 flex gap-3 items-start">

   583→  { key: 'PENDING_CONFIRM', label: 'Comanda enviada', icon: ClipboardList },
   584→  { key: 'IN_KITCHEN', label: 'Na cozinha', icon: ChefHat },

   589→function itemProgress(status: string): number {
   590→  const idx = ITEM_FLOW.indexOf(status)
   591→  return ((idx + 1) / ITEM_FLOW.length) * 100
   592→}

   594→function TrackingPhase({ order, onFinish, finishing }: { order: ClientOrder; onFinish: () => void; finishing: boolean }) {
   595→  const [now, setNow] = useState(Date.now())
   596→
   597→  useEffect(() => {
   598→    const t = setInterval(() => setNow(Date.now()), 1000)
   599→    return () => clearInterval(t)
   600→  }, [])
   601→
   602→  const elapsedMin = Math.floor((now - new Date(order.createdAt).getTime()) / 60000)
   603→  const servedCount = order.items.filter((i) => i.status === 'SERVED').length
   604→  const readyCount = order.items.filter((i) => i.status === 'READY').length
   605→  const allServed = servedCount === order.items.length
   606→
   607→  return (
   608→    <div className="apex-enter space-y-4">
   609→      <Card className="border-primary/30 apex-gradient-soft">
   610→        <CardContent className="p-4">
   611→          <div className="flex items-center justify-between">
   612→            <div>
   613→              <p className="text-xs text-muted-foreground">Comanda {order.code}</p>
   614→              <p className="font-bold text-lg">Garçom: {order.waiterName ?? 'Atribuindo…'}</p>
   615→            </div>
   616→            <div className="text-right">
   617→              <p className="text-xs text-muted-foreground">Tempo total</p>
   618→              <p className="font-bold text-lg text-primary">{elapsedMin} min</p>
   619→            </div>
   620→          </div>
   621→          <div className="mt-3 flex gap-1">
   622→            {order.items.map((i) => (
   623→              <div key={i.id} className="flex-1 flex gap-1">
   624→                {Array.from({ length: i.quantity }).map((_, q) => (
   625→                  <div key={q} className="h-1.5 flex-1 rounded-full bg-muted overflow-hidden">
   626→                    <div
   627→                      className={cn('h-full rounded-full transition-all duration-500', {
   628→                        'bg-zinc-400 dark:bg-zinc-600': i.status === 'PENDING',
   629→                        'bg-amber-500': i.status === 'IN_PREPARATION',
   630→                        'bg-emerald-500': i.status === 'READY' || i.status === 'SERVED',
   631→                      })}
   632→                      style={{ width: i.status === 'PENDING' ? '25%' : i.status === 'IN_PREPARATION' ? '60%' : '100%' }}
   633→                    />
   634→                  </div>
   635→                ))}
   636→              </div>
   637→            ))}
   638→          </div>
   639→          <p className="text-[11px] text-muted-foreground mt-2">
   640→            {readyCount > 0 && !allServed
   641→              ? `${readyCount} prato(s) pronto(s) — o garçom já foi avisado!`
   642→              : allServed
   643→                ? 'Tudo servido. Bom apetite!'
   644→                : 'Seus pratos estão em andamento na cozinha.'}
   645→          </p>
   646→        </CardContent>
   647→      </Card>
   648→
   649→      {order.items.map((item) => {
   650→        const stepIdx = item.status === 'PENDING' ? 0 : item.status === 'IN_PREPARATION' ? 1 : item.status === 'READY' ? 2 : 3
   651→        const started = item.startedAt ? new Date(item.startedAt).getTime() : null
   652→        const elapsedSec = started ? Math.floor((now - started) / 1000) : 0
   653→        const remaining = Math.max(0, item.prepTime * 60 - elapsedSec)
   654→        const overdue = item.status === 'IN_PREPARATION' && remaining === 0
   655→        const pct = itemProgress(item.status)
   656→
   657→        return (
   658→          <Card key={item.id} className={cn(overdue && 'border-red-500/50')}>
   659→            <CardContent className="p-4 space-y-3">
   660→              <div className="flex items-start justify-between gap-2">
   661→                <div className="flex items-center gap-3">
   662→                  <span className="text-2xl">{item.emoji}</span>
   663→                  <div>
   664→                    <p className="text-sm font-semibold leading-tight">
   665→                      {item.quantity}× {item.productName}
   666→                    </p>
   667→                    {item.notes && <p className="text-[11px] text-amber-600 dark:text-amber-400 mt-0.5">“{item.notes}”</p>}
   668→                  </div>
   669→                </div>
   670→                <Badge
   671→                  variant="outline"
   672→                  className={cn('text-[11px]', {
   673→                    'border-zinc-400/50 text-muted-foreground': item.status === 'PENDING',
   674→                    'border-amber-500/50 text-amber-600 dark:text-amber-400': item.status === 'IN_PREPARATION',
   675→                    'border-emerald-500/50 text-emerald-600 dark:text-emerald-400': item.status === 'READY',
   676→                    'border-primary/50 text-primary': item.status === 'SERVED',
   677→                  })}
   678→                >
   679→                  {ITEM_STATUS_LABELS[item.status]}
   680→                </Badge>
   681→              </div>
   682→
   683→              {/* Fluxo de status */}
   684→              <div className="flex items-center justify-between px-0.5">
   685→                {STATUS_STEPS.map((s, i) => {
   686→                  const Icon = s.icon
   687→                  const done = i <= stepIdx
   688→                  return (
   689→                    <div key={s.key} className="flex flex-col items-center gap-1 flex-1">
   690→                      <div className="flex items-center w-full">
   691→                        <div className={cn('h-0.5 flex-1', i === 0 ? 'opacity-0' : done ? 'bg-primary' : 'bg-border')} />
   692→                        <Icon className={cn('h-4 w-4 shrink-0 transition-colors', done ? 'text-primary' : 'text-muted-foreground/40')} />
   693→                        <div className={cn('h-0.5 flex-1', i === STATUS_STEPS.length - 1 ? 'opacity-0' : i < stepIdx ? 'bg-primary' : 'bg-border')} />
   694→                      </div>
   695→                      <span className={cn('text-[9px] leading-none text-center', done ? 'text-foreground font-medium' : 'text-muted-foreground/50')}>
   696→                        {s.label}
   697→                      </span>
   698→                    </div>
   699→                  )
   700→                })}
   701→              </div>
   702→
   703→              <div className="flex items-center gap-3">
   704→                <Progress value={pct} className="h-1.5" />
   705→                {item.status === 'PENDING' && (
   706→                  <span className="text-[11px] text-muted-foreground shrink-0">aguardando cozinha</span>
   707→                )}
   708→                {item.status === 'IN_PREPARATION' && (
   709→                  <span className={cn('text-[11px] shrink-0 font-mono tabular-nums', overdue ? 'text-red-500 font-bold' : 'text-amber-600 dark:text-amber-400 font-medium')}>
   710→                    {overdue ? 'no ponto!' : `${Math.floor(remaining / 60)}:${String(remaining % 60).padStart(2, '0')}`}
   711→                  </span>
   712→                )}
   713→                {(item.status === 'READY' || item.status === 'SERVED') && (
   714→                  <span className="text-[11px] text-emerald-600 dark:text-emerald-400 shrink-0 font-medium">concluído</span>
   715→                )}
   716→              </div>
   717→            </CardContent>
   718→          </Card>
   719→        )
   720→      })}
   721→
   722→      <Card>
   723→        <CardContent className="p-4">
   724→          <div className="flex items-center justify-between text-sm">
   725→            <span className="text-muted-foreground">Consumo até agora</span>
   726→            <span className="font-bold text-primary">{currency(order.total)}</span>
   727→          </div>
   728→        </CardContent>
   729→      </Card>
   730→
   731→      <Button
   732→        onClick={onFinish}
   733→        disabled={finishing}
   734→        className="w-full h-12 text-base apex-gradient text-white font-semibold"
   735→      >
   736→        {finishing ? <Loader2 className="h-5 w-5 animate-spin" /> : <CheckCircle2 className="h-5 w-5" />}
   737→        Concluir consumo — enviar ao caixa
   738→      </Button>
   739→      <p className="text-center text-[11px] text-muted-foreground">
   740→        A comanda será finalizada e encaminhada ao caixa para pagamento.
   741→      </p>
   742→    </div>
   743→  )
   744→}
