    52→    52→  const [phase, setPhase] = useState<0 | 1 | 2 | 3 | 4>(0)
    53→    53→  const [cart, setCart] = useState<CartLine[]>([])

    55→    55→  const [tick, setTick] = useState(0)

    59→    59→    queryKey: ['client', token, tick],
    60→    60→    queryFn: () => api<ClientData>(`/api/client/${token}`),
    61→    61→    refetchInterval: 3000,
    62→    62→    retry: 2,
    63→    63→  })
    64→    64→
    65→    65→  const productsMap = useMemo(() => {

    70→    70→
    71→    71→  const orderId = data?.order?.id
    72→    72→
    73→    73→  // Realtime — sala da comanda do cliente
    74→    74→  useEffect(() => {
    75→    75→    const socket = getSharedSocket()

    77→    77→    socket.emit('join', { clientRoom: `client:${orderId}` })

    79→    79→    socket.on('item:atualizado', onChange)
    80→    80→    socket.on('comanda:paga', onChange)
    81→    81→    socket.on('comanda:confirmada', onChange)

    83→    83→      socket.off('item:atualizado', onChange)
    84→    84→      socket.off('comanda:paga', onChange)
    85→    85→      socket.off('comanda:confirmada', onChange)
    86→    86→    }
    87→    87→  }, [orderId])
    88→    88→

    90→    90→  const liveOrder = data?.order ?? null
    91→    91→  const displayPhase: 0 | 1 | 2 | 3 | 4 =
    92→    92→    liveOrder?.status === 'AWAITING_PAYMENT'
    93→    93→      ? 4
    94→    94→      : !liveOrder && data?.lastPaid && phase >= 3
    95→    95→        ? 4
    96→    96→        : phase
    97→    97→
    98→    98→  const finish = useMutation({
    99→    99→    mutationFn: () => apiPost(`/api/client/${token}`, { action: 'finish', orderId }),
   100→   100→    onSuccess: () => {
   101→   101→      playSound('success')
   102→   102→      toast.success('Comanda enviada ao caixa!')
   103→   103→      void refetch()
   104→   104→    },
   105→   105→    onError: (e: Error) => toast.error(e.message),
   106→   106→  })
   107→   107→
   108→   108→  if (isLoading) {
   109→   109→    return (
   110→   110→      <div className="min-h-screen bg-background flex items-center justify-center">
   111→   111→        <div className="flex items-center gap-2 text-muted-foreground text-sm">
   112→   112→          <Loader2 className="h-5 w-5 animate-spin" /> Conectando à mesa…
   113→   113→        </div>
   114→   114→      </div>
   115→   115→    )
   116→   116→  }
   117→   117→
   118→   118→  if (isError || !data) {
   119→   119→    return (
   120→   120→      <div className="min-h-screen bg-background flex items-center justify-center p-6">
   121→   121→        <Card className="max-w-sm w-full text-center p-8">
   122→   122→          <p className="text-4xl">🚨</p>
   123→   123→          <p className="font-semibold mt-3">Mesa indisponível</p>
   124→   124→          <p className="text-sm text-muted-foreground mt-1">Não foi possível identificar o QR Code. Chame um funcionário.</p>
   125→   125→          <Button className="mt-6" variant="outline" onClick={onExit}>Painel da equipe</Button>
   126→   126→        </Card>
   127→   127→      </div>
   128→   128→    )
   129→   129→  }
   130→   130→
   131→   131→  const cartCount = cart.reduce((a, c) => a + c.quantity, 0)
   132→   132→  const cartSum = cart.reduce((a, c) => a + (productsMap.get(c.productId)?.price ?? 0) * c.quantity, 0)
   133→   133→
   134→   134→  return (
   135→   135→    <div className="min-h-screen bg-background flex flex-col">
   136→   136→      <header className="sticky top-0 z-20 apex-gradient text-white shadow-lg">
   137→   137→        <div className="max-w-md mx-auto px-4 h-14 flex items-center gap-3">
   138→   138→          <div className="h-8 w-8 rounded-lg bg-white/15 ring-1 ring-white/25 flex items-center justify-center">
   139→   139→            <img src="/apex-logo.png" alt="Logo APEX FOOD" className="h-5 w-5 object-contain" />
   140→   140→          </div>
   141→   141→          <div className="flex-1 min-w-0">
   142→   142→            <p className="text-sm font-bold leading-none">Mesa {String(data.table.number).padStart(2, '0')}</p>
   143→   143→            <p className="text-[11px] text-white/75 mt-0.5">APEX FOOD</p>
   144→   144→          </div>
   145→   145→          {phase === 1 && cartCount > 0 && (
   146→   146→            <Button size="sm" variant="ghost" className="text-white hover:bg-white/15 h-8" onClick={() => setPhase(2)}>
   147→   147→              <ShoppingBag className="h-4 w-4 mr-1" /> {cartCount}
   148→   148→            </Button>
   149→   149→          )}
   150→   150→        </div>
   151→   151→        <div className="max-w-md mx-auto px-4 pb-2.5">
   152→   152→          <div className="flex gap-1">
   153→   153→            {STEPS.map((_, i) => (
   154→   154→              <div key={i} className={cn('h-1.5 flex-1 rounded-full transition-colors duration-300', i <= displayPhase ? 'bg-white' : 'bg-white/25')} />
   155→   155→            ))}
   156→   156→          </div>
   157→   157→          <div className="flex justify-between mt-1.5">
   158→   158→            {STEPS.map((s, i) => (
   159→   159→              <span key={s} className={cn('text-[9px] leading-none', i === displayPhase ? 'text-white font-semibold' : 'text-white/55')}>
   160→   160→                {s}
   161→   161→              </span>
   162→   162→            ))}
   163→   163→          </div>
   164→   164→        </div>
   165→   165→      </header>
   166→   166→
   167→   167→      <main className="flex-1 max-w-md w-full mx-auto p-4 pb-32">
   168→   168→        {phase === 0 && (
   169→   169→          <WelcomePhase tableNumber={data.table.number} hasOpen={!!data.order} onStart={() => (data.order ? setPhase(3) : setPhase(1))} />
   170→   170→        )}
   171→   171→        {phase === 1 && (
   172→   172→          <MenuPhase
   173→   173→            productsMap={productsMap}
   174→   174→            menu={menu.data?.categories ?? []}
   175→   175→            activeCat={activeCat}
   176→   176→            setActiveCat={setActiveCat}
   177→   177→            cart={cart}
   178→   178→            onAdd={(line) => {
   179→   179→              setCart((prev) => {
   180→   180→                const existing = prev.find((c) => c.productId === line.productId)
   181→   181→                if (existing) {
   182→   182→                  return prev.map((c) => (c.productId === line.productId ? { ...line, quantity: c.quantity + line.quantity } : c))
   183→   183→                }
   184→   184→                return [...prev, line]
   185→   185→              })
   186→   186→            }}
   187→   187→          />
   188→   188→        )}
   189→   189→        {phase === 2 && (
   190→   190→          <ReviewPhase
   191→   191→            cart={cart}
   192→   192→            productsMap={productsMap}
   193→   193→            onCartChange={setCart}
   194→   194→            onBack={() => setPhase(1)}
   195→   195→            onSent={() => {
   196→   196→              setCart([])
   197→   197→              setPhase(3)
   198→   198→              void refetch()
   199→   199→            }}
   200→   200→            tableToken={token}
   201→   201→          />
   202→   202→        )}
   203→   203→        {displayPhase === 3 && liveOrder && (
   204→   204→          <TrackingPhase order={liveOrder} onFinish={() => finish.mutate()} finishing={finish.isPending} />
   205→   205→        )}
   206→   206→        {displayPhase === 4 && (
   207→   207→          <ClosingPhase data={data} tick={tick} onNew={() => { void refetch(); setPhase(1) }} />
   208→   208→        )}
   209→   209→      </main>
   210→   210→
   211→   211→      {(phase === 1 || phase === 2) && (
   212→   212→        <div className="fixed bottom-0 inset-x-0 border-t bg-background/92 backdrop-blur p-3 z-20">
   213→   213→          <div className="max-w-md mx-auto flex items-center gap-2">
   214→   214→            {phase === 2 && (
   215→   215→              <Button variant="outline" onClick={() => setPhase(1)} className="h-11">
   216→   216→                <ChevronLeft className="h-4 w-4" /> Cardápio
   217→   217→              </Button>
   218→   218→            )}
   219→   219→            {phase === 1 && (
   220→   220→              <Button
   221→   221→                onClick={() => setPhase(2)}
   222→   222→                disabled={cart.length === 0}
   223→   223→                className="flex-1 h-11 apex-gradient text-white font-semibold"
   224→   224→              >
   225→   225→                <ClipboardList className="h-4 w-4" />
   226→   226→                Revisar comanda
   227→   227→                <span className="mx-1">·</span> {currency(cartSum)}

   229→   229→              </Button>
   230→   230→            )}
   231→   231→          </div>
   232→   232→        </div>
   233→   233→      )}
   234→   234→
   235→   235→      <p className="pb-4 text-center text-[10px] text-muted-foreground">
   236→   236→        <button className="underline" onClick={onExit}>Painel da equipe</button>
   237→   237→      </p>
   238→   238→    </div>
   239→   239→  )
   240→   240→}

   275→   275→function MenuPhase({
   276→   276→  menu, activeCat, setActiveCat, cart, onAdd, productsMap,
   277→   277→}: {
   278→   278→  menu: Array<Category & { products: Product[] }>
   279→   279→  activeCat: string | null
   280→   280→  setActiveCat: (c: string | null) => void
   281→   281→  cart: CartLine[]
   282→   282→  onAdd: (line: CartLine) => void
   283→   283→  productsMap: Map<string, Product>
   284→   284→}) {
   285→   285→  const current = useMemo(() => menu.find((c) => c.id === activeCat) ?? menu[0], [menu, activeCat])
   286→   286→
   287→   287→  if (menu.length === 0) {
   288→   288→    return <div className="text-center text-muted-foreground text-sm py-16">Cardápio em atualização…</div>
   289→   289→  }
   290→   290→
   291→   291→  const countInCart = (id: string) => cart.find((c) => c.productId === id)?.quantity ?? 0
   292→   292→
   293→   293→  return (
   294→   294→    <div className="apex-enter">
   295→   295→      <div className="flex gap-1.5 overflow-x-auto pb-2 -mx-1 px-1">
   296→   296→        {menu.map((c) => (
   297→   297→          <button
   298→   298→            key={c.id}
   299→   299→            onClick={() => setActiveCat(c.id)}
   300→   300→            className={cn(
   301→   301→              'shrink-0 rounded-full border px-3.5 py-2 text-xs font-medium transition-all flex items-center gap-1.5',
   302→   302→              current?.id === c.id
   303→   303→                ? 'border-primary/50 bg-primary/10 text-primary'
   304→   304→                : 'bg-card text-muted-foreground hover:text-foreground'
   305→   305→            )}
   306→   306→          >
   307→   307→            <span>{c.icon}</span> {c.name}
   308→   308→          </button>
   309→   309→        ))}
   310→   310→      </div>
   311→   311→
   312→   312→      <div className="mt-4 space-y-3">
   313→   313→        {current?.products.map((p) => (
   314→   314→          <ProductCard key={p.id} product={p} inCart={countInCart(p.id)} onAdd={onAdd} />
   315→   315→        ))}
   316→   316→        {current?.products.length === 0 && (
   317→   317→          <p className="text-center text-sm text-muted-foreground py-10">Sem itens disponíveis nesta categoria.</p>
   318→   318→        )}
   319→   319→      </div>
   320→   320→
   321→   321→      <div className="text-center text-[10px] text-muted-foreground mt-6">
   322→   322→        {productsMap.size} itens no cardápio · preços em reais
   323→   323→      </div>
   324→   324→    </div>
   325→   325→  )
   326→   326→}
