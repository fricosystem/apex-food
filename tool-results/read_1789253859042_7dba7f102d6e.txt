    59→/** Número com animação count-up (ease-out cúbico), respeitando prefers-reduced-motion */
    60→function AnimatedNumber({ value, format }: { value: number; format?: (v: number) => string }) {
    61→  const [display, setDisplay] = useState(value)
    62→  const prevRef = useRef(value)
    63→
    64→  useEffect(() => {
    65→    const from = prevRef.current
    66→    const to = value
    67→    prevRef.current = value
    68→    if (from === to) return
    69→    if (typeof window !== 'undefined' && window.matchMedia?.('(prefers-reduced-motion: reduce)').matches) {
    70→      const id = requestAnimationFrame(() => setDisplay(to))
    71→      return () => cancelAnimationFrame(id)
    72→    }
    73→    const duration = 900
    74→    let raf = 0
    75→    const t0 = performance.now()
    76→    const tick = (t: number) => {
    77→      const p = Math.min(1, (t - t0) / duration)
    78→      const eased = 1 - Math.pow(1 - p, 3)
    79→      setDisplay(from + (to - from) * eased)
    80→      if (p < 1) raf = requestAnimationFrame(tick)
    81→    }
    82→    raf = requestAnimationFrame(tick)
    83→    return () => cancelAnimationFrame(raf)
    84→  }, [value])
    85→
    86→  return <>{format ? format(display) : Math.round(display).toLocaleString('pt-BR')}</>
    87→}

    89→/** Badge de variação vs. período anterior (verde ↑ / vermelho ↓) */
    90→function DeltaBadge({ current, previous }: { current: number; previous: number }) {
    91→  if (!previous || previous <= 0) {
    92→    if (current > 0) {
    93→      return (
    94→        <span className="inline-flex items-center gap-1 rounded-full px-2 py-0.5 text-[10px] font-semibold border bg-emerald-500/10 text-emerald-600 dark:text-emerald-400 border-emerald-500/30">
    95→          <TrendingUp className="h-3 w-3" /> novo
    96→        </span>
    97→      )
    98→    }
    99→    return null
   100→  }
   101→  const pct = ((current - previous) / previous) * 100
   102→  const up = pct >= 0
   103→  return (
   104→    <span
   105→      className={cn(
   106→        'inline-flex items-center gap-1 rounded-full px-2 py-0.5 text-[10px] font-semibold border',
   107→        up
   108→          ? 'bg-emerald-500/10 text-emerald-600 dark:text-emerald-400 border-emerald-500/30'
   109→          : 'bg-red-500/10 text-red-600 dark:text-red-400 border-red-500/30',
   110→      )}
   111→    >
   112→      {up ? <TrendingUp className="h-3 w-3" /> : <TrendingDown className="h-3 w-3" />}
   113→      {up ? '+' : ''}{pct.toFixed(1)}%
   114→    </span>
   115→  )
   116→}
