   108→const clamp01 = (v: number) => Math.min(1, Math.max(0, v))
   109→
   110→/**
   111→ * Fade vinculado à posição de rolagem (sem transição CSS — o estilo é recalculado a cada frame):
   112→ * - rolando para baixo: o que entra por baixo sofre fade in subindo (+44px → 0) e o que
   113→ *   sai por cima sofre fade out (0 → -30px);
   114→ * - rolando para cima (reverse): o mesmo cálculo posicional inverte o efeito — o conteúdo
   115→ *   superior volta com fade in descendo e o inferior some com fade out.
   116→ */
   117→type FadeApi = { register: (el: HTMLDivElement | null) => void }
   118→const FadeCtx = createContext<FadeApi | null>(null)
   119→
   120→function useFadeController(): FadeApi {
   121→  const items = useRef(new Set<HTMLDivElement>())
   122→  const rafRef = useRef(0)
   123→
   124→  const update = useCallback(() => {
   125→    rafRef.current = 0
   126→    const H = window.innerHeight || 1
   127→    const reduced = window.matchMedia('(prefers-reduced-motion: reduce)').matches
   128→    items.current.forEach((el) => {
   129→      if (!el.isConnected) {
   130→        items.current.delete(el)
   131→        return
   132→      }
   133→      if (reduced) {
   134→        el.style.opacity = '1'
   135→        el.style.transform = ''
   136→        return
   137→      }
   138→      const rect = el.getBoundingClientRect()
   139→      // Desconta o transform aplicado no frame anterior para medir a posição de layout
   140→      const prevY = Number(el.dataset.fadeY ?? 0)
   141→      const top = rect.top - prevY
   142→      const bottom = rect.bottom - prevY
   143→      // Entrada: topo percorrendo os 22% inferiores da viewport
   144→      const tIn = clamp01((H - top) / (H * 0.22))
   145→      // Saída: base adentrando os 28% superiores da viewport
   146→      const tOut = clamp01(bottom / (H * 0.28))
   147→      // Blocos ancorados na primeira dobra nascem visíveis e sofrem apenas o fade de saída
   148→      const anchored = el.dataset.fadeAnchored === '1'
   149→      const opacity = anchored ? tOut : Math.min(tIn, tOut)
   150→      const y = (anchored ? 0 : (1 - tIn) * 44) - (1 - tOut) * 30
   151→      el.dataset.fadeY = y.toFixed(1)
   152→      el.style.opacity = opacity.toFixed(3)
   153→      el.style.transform = `translate3d(0, ${y.toFixed(1)}px, 0)`
   154→    })
   155→  }, [])
   156→
   157→  const schedule = useCallback(() => {
   158→    if (!rafRef.current) rafRef.current = requestAnimationFrame(update)
   159→  }, [update])
   160→
   161→  const register = useCallback((el: HTMLDivElement | null) => {
   162→    if (el) items.current.add(el)
   163→  }, [])
   164→
   165→  useEffect(() => {
   166→    schedule()
   167→    window.addEventListener('scroll', schedule, { passive: true })
   168→    window.addEventListener('resize', schedule)
   169→    // Recalcula após fontes/imagens estabilizarem o layout
   170→    const t1 = window.setTimeout(schedule, 350)
   171→    const t2 = window.setTimeout(schedule, 900)
   172→    document.fonts?.ready.then(schedule).catch(() => {})
   173→    return () => {
   174→      window.removeEventListener('scroll', schedule)
   175→      window.removeEventListener('resize', schedule)
   176→      window.clearTimeout(t1)
   177→      window.clearTimeout(t2)
   178→      if (rafRef.current) cancelAnimationFrame(rafRef.current)
   179→    }
   180→  }, [schedule])
   181→
   182→  return { register }
   183→}

   185→/** Bloco que participa do fade de rolagem; startVisible mantém o bloco visível antes da hidratação e anchored desativa o fade de entrada (só sai por cima) */
   186→function ScrollFade({ children, className, startVisible, anchored }: {
   187→  children: React.ReactNode
   188→  className?: string
   189→  startVisible?: boolean
   190→  anchored?: boolean
   191→}) {
   192→  const fade = useContext(FadeCtx)
   193→  return (
   194→    <div
   195→      ref={fade?.register}
   196→      className={cn('relative', className)}
   197→      data-fade-anchored={anchored ? '1' : undefined}
   198→      data-fade-y={startVisible ? undefined : '44'}
   199→      style={{
   200→        opacity: startVisible ? undefined : 0,
   201→        transform: startVisible ? undefined : 'translate3d(0, 44px, 0)',
   202→        willChange: 'opacity, transform',
   203→      }}
   204→    >
   205→      {children}
   206→    </div>
   207→  )
   208→}
