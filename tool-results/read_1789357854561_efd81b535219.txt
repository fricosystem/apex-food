    59→function ProductsTab() {
    60→  const qc = useQueryClient()
    61→  const [editing, setEditing] = useState<Product | null>(null)
    62→  const [creating, setCreating] = useState(false)
    63→
    64→  const { data, isLoading } = useQuery<{ categories: Array<Category & { products: Product[] }> }>({
    65→    queryKey: ['categories', 'all'],
    66→    queryFn: () => api('/api/categories?all=1'),
    67→  })
    68→
    69→  const toggle = useMutation({
    70→    mutationFn: (p: Product) => apiPatch(`/api/products/${p.id}`, { active: !p.active }),
    71→    onSuccess: () => {
    72→      void qc.invalidateQueries({ queryKey: ['categories'] })
    73→      void qc.invalidateQueries({ queryKey: ['products'] })
    74→      toast.success('Produto atualizado')
    75→    },
    76→    onError: (e: Error) => toast.error(e.message),
    77→  })
    78→  const remove = useMutation({
    79→    mutationFn: (id: string) => apiDelete(`/api/products/${id}`),
    80→    onSuccess: () => {
    81→      void qc.invalidateQueries({ queryKey: ['categories'] })
    82→      toast.success('Produto removido (ou desativado, se já possui histórico)')
    83→    },
    84→    onError: (e: Error) => toast.error(e.message),
    85→  })
    86→
    87→  const allProducts = (data?.categories ?? []).flatMap((c) => c.products)
    88→
    89→  return (
    90→    <div className="space-y-3">
    91→      <div className="flex items-center justify-between">
    92→        <p className="text-sm text-muted-foreground">{allProducts.filter((p) => p.active).length} produtos ativos · o tempo de preparo alimenta a cozinha e o cliente</p>
    93→        <Button onClick={() => setCreating(true)} className="apex-gradient text-white">
    94→          <Plus className="h-4 w-4" /> Novo produto
    95→        </Button>
    96→      </div>
    97→
    98→      {isLoading ? (
    99→        <Skeleton className="h-72 rounded-xl" />
   100→      ) : (
   101→        <div className="grid gap-3 md:grid-cols-2 xl:grid-cols-3">
   102→          {allProducts.map((p) => (
   103→            <Card key={p.id} className={cn(!p.active && 'opacity-55')}>
   104→              <CardContent className="p-4 flex gap-3">
   105→                {p.image ? (
   106→                  <img src={p.image} alt={p.name} className="h-16 w-16 rounded-lg object-cover shrink-0" />
   107→                ) : (
   108→                  <div className="h-16 w-16 rounded-lg apex-gradient-soft flex items-center justify-center text-3xl shrink-0">{p.emoji}</div>
   109→                )}
   110→                <div className="flex-1 min-w-0">
   111→                  <div className="flex items-start justify-between gap-2">
   112→                    <p className="font-semibold text-sm leading-tight truncate">{p.name}</p>
   113→                    <Switch checked={p.active} onCheckedChange={() => toggle.mutate(p)} aria-label="Ativar produto" />
   114→                  </div>
   115→                  <p className="text-[11px] text-muted-foreground mt-0.5">{p.category.icon} {p.category.name}</p>
   116→                  <p className="text-sm font-bold text-primary mt-1">{currency(p.price)}</p>
   117→                  <div className="flex items-center gap-2 mt-1.5">
   118→                    <Badge variant="outline" className="text-[10px] gap-1"><Clock className="h-3 w-3" /> {p.prepTime} min</Badge>
   119→                    <button className="text-[11px] text-muted-foreground hover:text-foreground flex items-center gap-1" onClick={() => setEditing(p)}>
   120→                      <Pencil className="h-3 w-3" /> editar
   121→                    </button>
   122→                    <button className="text-[11px] text-red-500 hover:text-red-400 flex items-center gap-1" onClick={() => remove.mutate(p.id)}>
   123→                      <Trash2 className="h-3 w-3" /> excluir
   124→                    </button>
   125→                  </div>
   126→                </div>
   127→              </CardContent>
   128→            </Card>
   129→          ))}
   130→        </div>
   131→      )}
   132→
   133→      <ProductDialog open={creating || !!editing} product={editing} onClose={() => { setCreating(false); setEditing(null) }} />
   134→    </div>
   135→  )
   136→}

   138→function ProductDialog({ open, product, onClose }: { open: boolean; product: Product | null; onClose: () => void }) {
   139→  const qc = useQueryClient()
   140→  const { data } = useQuery<{ categories: Category[] }>({
   141→    queryKey: ['categories', 'all'],
   142→    queryFn: () => api('/api/categories?all=1'),
   143→    enabled: open,
   144→  })
   145→
   146→  const [form, setForm] = useState({ name: '', description: '', price: '', prepTime: '15', emoji: '🍽️', image: '', categoryId: '' })
   147→  const [loadedFor, setLoadedFor] = useState<Product | null>(null)
   148→
   149→  if (open && product && loadedFor !== product) {
   150→    setLoadedFor(product)
   151→    setForm({
   152→      name: product.name, description: product.description, price: String(product.price),
   153→      prepTime: String(product.prepTime), emoji: product.emoji, image: product.image ?? '', categoryId: product.categoryId,
   154→    })
   155→  }
   156→  if (open && !product && loadedFor !== null) {
   157→    setLoadedFor(null)
   158→    setForm({ name: '', description: '', price: '', prepTime: '15', emoji: '🍽️', image: '', categoryId: '' })
   159→  }
   160→
   161→  const save = useMutation({
   162→    mutationFn: () => {
   163→      const payload = {
   164→        name: form.name,
   165→        description: form.description,
   166→        price: Number(form.price.replace(',', '.')),
   167→        prepTime: Number(form.prepTime),
   168→        emoji: form.emoji || '🍽️',
   169→        image: form.image || null,
   170→        categoryId: form.categoryId,
   171→      }
   172→      return product ? apiPatch(`/api/products/${product.id}`, payload) : apiPost('/api/products', payload)
   173→    },
   174→    onSuccess: () => {
   175→      void qc.invalidateQueries({ queryKey: ['categories'] })
   176→      void qc.invalidateQueries({ queryKey: ['products'] })
   177→      toast.success(product ? 'Produto atualizado' : 'Produto criado')
   178→      onClose()
   179→    },
   180→    onError: (e: Error) => toast.error(e.message),
   181→  })
   182→
   183→  return (
   184→    <Dialog open={open} onOpenChange={(v) => !v && onClose()}>
   185→      <DialogContent className="max-w-md max-h-[90vh] overflow-y-auto">
   186→        <DialogHeader>
   187→          <DialogTitle>{product ? 'Editar produto' : 'Novo produto'}</DialogTitle>
   188→        </DialogHeader>
   189→        <div className="space-y-3.5">
   190→          <div className="grid grid-cols-2 gap-3">
   191→            <div className="space-y-1.5">
   192→              <Label>Nome</Label>
   193→              <Input value={form.name} onChange={(e) => setForm({ ...form, name: e.target.value })} placeholder="Ex: Pizza Margherita" />
   194→            </div>
   195→            <div className="space-y-1.5">
   196→              <Label>Categoria</Label>
   197→              <Select value={form.categoryId} onValueChange={(v) => setForm({ ...form, categoryId: v })}>
   198→                <SelectTrigger><SelectValue placeholder="Selecione" /></SelectTrigger>
   199→                <SelectContent>
   200→                  {(data?.categories ?? []).map((c) => (
   201→                    <SelectItem key={c.id} value={c.id}>{c.icon} {c.name}</SelectItem>
   202→                  ))}
   203→                </SelectContent>
   204→              </Select>
   205→            </div>
   206→          </div>
   207→          <div className="space-y-1.5">
   208→            <Label>Descrição</Label>
   209→            <Textarea value={form.description} onChange={(e) => setForm({ ...form, description: e.target.value })} className="min-h-[60px] text-sm" placeholder="Ingredientes e diferenciais" />
   210→          </div>
   211→          <div className="grid grid-cols-3 gap-3">
   212→            <div className="space-y-1.5">
   213→              <Label>Preço (R$)</Label>
   214→              <Input value={form.price} onChange={(e) => setForm({ ...form, price: e.target.value })} placeholder="49.90" inputMode="decimal" />
   215→            </div>
   216→            <div className="space-y-1.5">
   217→              <Label>Preparo (min)</Label>
   218→              <Input value={form.prepTime} onChange={(e) => setForm({ ...form, prepTime: e.target.value.replace(/\D/g, '') })} inputMode="numeric" />
   219→            </div>
   220→            <div className="space-y-1.5">
   221→              <Label>Emoji</Label>
   222→              <Input value={form.emoji} onChange={(e) => setForm({ ...form, emoji: e.target.value.slice(0, 2) })} />
   223→            </div>
   224→          </div>
   225→          <div className="space-y-1.5">
   226→            <Label className="flex items-center gap-1.5"><ImageIcon className="h-3.5 w-3.5" /> URL da imagem (opcional)</Label>
   227→            <Input value={form.image} onChange={(e) => setForm({ ...form, image: e.target.value })} placeholder="https://…" />
   228→          </div>
   229→        </div>
   230→        <DialogFooter>
   231→          <Button className="w-full apex-gradient text-white font-semibold" onClick={() => save.mutate()} disabled={save.isPending || !form.name || !form.categoryId}>
   232→            {save.isPending && <Loader2 className="h-4 w-4 animate-spin" />} Salvar produto
   233→          </Button>
   234→        </DialogFooter>
   235→      </DialogContent>
   236→    </Dialog>
   237→  )
   238→}

   420→      void qc.invalidateQueries({ queryKey: ['goals'] })
   421→      toast.success('Meta criada')
   422→      setOpen(false)
   423→    },
   424→    onError: (e: Error) => toast.error(e.message),
   425→  })
   426→  const remove = useMutation({
   427→    mutationFn: (id: string) => apiDelete(`/api/goals/${id}`),
   428→    onSuccess: () => {
   429→      void qc.invalidateQueries({ queryKey: ['goals'] })
   430→      toast.success('Meta removida')
   431→    },
   432→  })
   433→
   434→  return (
   435→    <div className="space-y-3">
   436→      <div className="flex items-center justify-between">
   437→        <p className="text-sm text-muted-foreground">Metas mensais de comandas por funcionário</p>
   438→        <Button onClick={() => setOpen(true)} className="apex-gradient text-white"><Plus className="h-4 w-4" /> Nova meta</Button>
   439→      </div>
   440→      {isLoading ? (
   441→        <Skeleton className="h-48 rounded-xl" />
   442→      ) : (data?.goals ?? []).length === 0 ? (
   443→        <Card><CardContent className="p-10 text-center text-sm text-muted-foreground">Nenhuma meta cadastrada para o mês atual.</CardContent></Card>
   444→      ) : (
   445→        <div className="grid gap-3 md:grid-cols-2 xl:grid-cols-3">
   446→          {(data?.goals ?? []).map((g) => {
   447→            const pct = Math.min(100, Math.round((g.achieved / Math.max(g.target, 1)) * 100))
   448→            return (
   449→              <Card key={g.id}>
   450→                <CardContent className="p-4 space-y-2.5">
   451→                  <div className="flex items-center justify-between">
   452→                    <p className="font-semibold text-sm">{g.user.name}</p>
   453→                    <button className="text-muted-foreground hover:text-red-500" onClick={() => remove.mutate(g.id)} aria-label="Excluir meta"><Trash2 className="h-3.5 w-3.5" /></button>
   454→                  </div>
   455→                  <p className="text-[11px] text-muted-foreground">{g.title} · {g.month}</p>
   456→                  <Progress value={pct} className="h-2" />
   457→                  <div className="flex items-center justify-between text-xs">
   458→                    <span className="text-muted-foreground">{g.achieved}/{g.target} comandas</span>
   459→                    <span className="font-semibold">{pct}%</span>
   460→                  </div>
   461→                </CardContent>
   462→              </Card>
   463→            )
   464→          })}
   465→        </div>
   466→      )}
   467→
   468→      <Dialog open={open} onOpenChange={setOpen}>
   469→        <DialogContent className="max-w-sm">
   470→          <DialogHeader><DialogTitle>Nova meta</DialogTitle></DialogHeader>
   471→          <div className="space-y-3">
   472→            <div className="space-y-1.5">
   473→              <Label>Funcionário</Label>
   474→              <Select value={form.userId} onValueChange={(v) => setForm({ ...form, userId: v })}>
   475→                <SelectTrigger><SelectValue placeholder="Selecione" /></SelectTrigger>
   476→                <SelectContent>
   477→                  {(usersData?.users ?? []).filter((u) => u.role === 'WAITER').map((u) => (
   478→                    <SelectItem key={u.id} value={u.id}>{u.name}</SelectItem>
   479→                  ))}
   480→                </SelectContent>
   481→              </Select>
   482→            </div>
   483→            <div className="space-y-1.5"><Label>Descrição</Label><Input value={form.title} onChange={(e) => setForm({ ...form, title: e.target.value })} /></div>
   484→            <div className="space-y-1.5"><Label>Alvo (comandas/mês)</Label><Input value={form.target} onChange={(e) => setForm({ ...form, target: e.target.value.replace(/\D/g, '') })} inputMode="numeric" /></div>
   485→          </div>
   486→          <DialogFooter>
   487→            <Button className="w-full apex-gradient text-white" onClick={() => save.mutate()} disabled={!form.userId || save.isPending}>Criar meta</Button>
   488→          </DialogFooter>
   489→        </DialogContent>
   490→      </Dialog>
   491→    </div>
   492→  )
   493→}
