    59→    59→/** Número com animação count-up (ease-out cúbico), respeitando prefers-reduced-motion */
    60→    60→function AnimatedNumber({ value, format }: { value: number; format?: (v: number) => string }) {
    61→    61→  const [display, setDisplay] = useState(value)
    62→    62→  const prevRef = useRef(value)
    63→    63→
    64→    64→  useEffect(() => {
    65→    65→    const from = prevRef.current
    66→    66→    const to = value
    67→    67→    prevRef.current = value
    68→    68→    if (from === to) return
    69→    69→    if (typeof window !== 'undefined' && window.matchMedia?.('(prefers-reduced-motion: reduce)').matches) {
    70→    70→      const id = requestAnimationFrame(() => setDisplay(to))
    71→    71→      return () => cancelAnimationFrame(id)
    72→    72→    }
    73→    73→    const duration = 900
    74→    74→    let raf = 0
    75→    75→    const t0 = performance.now()
    76→    76→    const tick = (t: number) => {
    77→    77→      const p = Math.min(1, (t - t0) / duration)
    78→    78→      const eased = 1 - Math.pow(1 - p, 3)
    79→    79→      setDisplay(from + (to - from) * eased)
    80→    80→      if (p < 1) raf = requestAnimationFrame(tick)
    81→    81→    }
    82→    82→    raf = requestAnimationFrame(tick)
    83→    83→    return () => cancelAnimationFrame(raf)
    84→    84→  }, [value])
    85→    85→
    86→    86→  return <>{format ? format(display) : Math.round(display).toLocaleString('pt-BR')}</>
    87→    87→}

    89→    89→/** Badge de variação vs. período anterior (verde ↑ / vermelho ↓) */
    90→    90→function DeltaBadge({ current, previous }: { current: number; previous: number }) {
    91→    91→  if (!previous || previous <= 0) {
    92→    92→    if (current > 0) {
    93→    93→      return (
    94→    94→        <span className="inline-flex items-center gap-1 rounded-full px-2 py-0.5 text-[10px] font-semibold border bg-emerald-500/10 text-emerald-600 dark:text-emerald-400 border-emerald-500/30">
    95→    95→          <TrendingUp className="h-3 w-3" /> novo
    96→    96→        </span>
    97→    97→      )
    98→    98→    }
    99→    99→    return null
   100→   100→  }
   101→   101→  const pct = ((current - previous) / previous) * 100
   102→   102→  const up = pct >= 0
   103→   103→  return (
   104→   104→    <span
   105→   105→      className={cn(
   106→   106→        'inline-flex items-center gap-1 rounded-full px-2 py-0.5 text-[10px] font-semibold border',
   107→   107→        up
   108→   108→          ? 'bg-emerald-500/10 text-emerald-600 dark:text-emerald-400 border-emerald-500/30'
   109→   109→          : 'bg-red-500/10 text-red-600 dark:text-red-400 border-red-500/30',
   110→   110→      )}
   111→   111→    >
   112→   112→      {up ? <TrendingUp className="h-3 w-3" /> : <TrendingDown className="h-3 w-3" />}
   113→   113→      {up ? '+' : ''}{pct.toFixed(1)}%
   114→   114→    </span>
   115→   115→  )
   116→   116→}
