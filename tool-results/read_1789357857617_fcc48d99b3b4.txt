    59→    59→function ProductsTab() {
    60→    60→  const qc = useQueryClient()
    61→    61→  const [editing, setEditing] = useState<Product | null>(null)
    62→    62→  const [creating, setCreating] = useState(false)
    63→    63→
    64→    64→  const { data, isLoading } = useQuery<{ categories: Array<Category & { products: Product[] }> }>({
    65→    65→    queryKey: ['categories', 'all'],
    66→    66→    queryFn: () => api('/api/categories?all=1'),
    67→    67→  })
    68→    68→
    69→    69→  const toggle = useMutation({
    70→    70→    mutationFn: (p: Product) => apiPatch(`/api/products/${p.id}`, { active: !p.active }),
    71→    71→    onSuccess: () => {
    72→    72→      void qc.invalidateQueries({ queryKey: ['categories'] })
    73→    73→      void qc.invalidateQueries({ queryKey: ['products'] })
    74→    74→      toast.success('Produto atualizado')
    75→    75→    },
    76→    76→    onError: (e: Error) => toast.error(e.message),
    77→    77→  })
    78→    78→  const remove = useMutation({
    79→    79→    mutationFn: (id: string) => apiDelete(`/api/products/${id}`),
    80→    80→    onSuccess: () => {
    81→    81→      void qc.invalidateQueries({ queryKey: ['categories'] })
    82→    82→      toast.success('Produto removido (ou desativado, se já possui histórico)')
    83→    83→    },
    84→    84→    onError: (e: Error) => toast.error(e.message),
    85→    85→  })
    86→    86→
    87→    87→  const allProducts = (data?.categories ?? []).flatMap((c) => c.products)
    88→    88→
    89→    89→  return (
    90→    90→    <div className="space-y-3">
    91→    91→      <div className="flex items-center justify-between">
    92→    92→        <p className="text-sm text-muted-foreground">{allProducts.filter((p) => p.active).length} produtos ativos · o tempo de preparo alimenta a cozinha e o cliente</p>
    93→    93→        <Button onClick={() => setCreating(true)} className="apex-gradient text-white">
    94→    94→          <Plus className="h-4 w-4" /> Novo produto
    95→    95→        </Button>
    96→    96→      </div>
    97→    97→
    98→    98→      {isLoading ? (
    99→    99→        <Skeleton className="h-72 rounded-xl" />
   100→   100→      ) : (
   101→   101→        <div className="grid gap-3 md:grid-cols-2 xl:grid-cols-3">
   102→   102→          {allProducts.map((p) => (
   103→   103→            <Card key={p.id} className={cn(!p.active && 'opacity-55')}>
   104→   104→              <CardContent className="p-4 flex gap-3">
   105→   105→                {p.image ? (
   106→   106→                  <img src={p.image} alt={p.name} className="h-16 w-16 rounded-lg object-cover shrink-0" />
   107→   107→                ) : (
   108→   108→                  <div className="h-16 w-16 rounded-lg apex-gradient-soft flex items-center justify-center text-3xl shrink-0">{p.emoji}</div>
   109→   109→                )}
   110→   110→                <div className="flex-1 min-w-0">
   111→   111→                  <div className="flex items-start justify-between gap-2">
   112→   112→                    <p className="font-semibold text-sm leading-tight truncate">{p.name}</p>
   113→   113→                    <Switch checked={p.active} onCheckedChange={() => toggle.mutate(p)} aria-label="Ativar produto" />
   114→   114→                  </div>
   115→   115→                  <p className="text-[11px] text-muted-foreground mt-0.5">{p.category.icon} {p.category.name}</p>
   116→   116→                  <p className="text-sm font-bold text-primary mt-1">{currency(p.price)}</p>
   117→   117→                  <div className="flex items-center gap-2 mt-1.5">
   118→   118→                    <Badge variant="outline" className="text-[10px] gap-1"><Clock className="h-3 w-3" /> {p.prepTime} min</Badge>
   119→   119→                    <button className="text-[11px] text-muted-foreground hover:text-foreground flex items-center gap-1" onClick={() => setEditing(p)}>
   120→   120→                      <Pencil className="h-3 w-3" /> editar
   121→   121→                    </button>
   122→   122→                    <button className="text-[11px] text-red-500 hover:text-red-400 flex items-center gap-1" onClick={() => remove.mutate(p.id)}>
   123→   123→                      <Trash2 className="h-3 w-3" /> excluir
   124→   124→                    </button>
   125→   125→                  </div>
   126→   126→                </div>
   127→   127→              </CardContent>
   128→   128→            </Card>
   129→   129→          ))}
   130→   130→        </div>
   131→   131→      )}
   132→   132→
   133→   133→      <ProductDialog open={creating || !!editing} product={editing} onClose={() => { setCreating(false); setEditing(null) }} />
   134→   134→    </div>
   135→   135→  )
   136→   136→}

   138→   138→function ProductDialog({ open, product, onClose }: { open: boolean; product: Product | null; onClose: () => void }) {
   139→   139→  const qc = useQueryClient()
   140→   140→  const { data } = useQuery<{ categories: Category[] }>({
   141→   141→    queryKey: ['categories', 'all'],
   142→   142→    queryFn: () => api('/api/categories?all=1'),
   143→   143→    enabled: open,
   144→   144→  })
   145→   145→
   146→   146→  const [form, setForm] = useState({ name: '', description: '', price: '', prepTime: '15', emoji: '🍽️', image: '', categoryId: '' })
   147→   147→  const [loadedFor, setLoadedFor] = useState<Product | null>(null)
   148→   148→
   149→   149→  if (open && product && loadedFor !== product) {
   150→   150→    setLoadedFor(product)
   151→   151→    setForm({
   152→   152→      name: product.name, description: product.description, price: String(product.price),
   153→   153→      prepTime: String(product.prepTime), emoji: product.emoji, image: product.image ?? '', categoryId: product.categoryId,
   154→   154→    })
   155→   155→  }
   156→   156→  if (open && !product && loadedFor !== null) {
   157→   157→    setLoadedFor(null)
   158→   158→    setForm({ name: '', description: '', price: '', prepTime: '15', emoji: '🍽️', image: '', categoryId: '' })
   159→   159→  }
   160→   160→
   161→   161→  const save = useMutation({
   162→   162→    mutationFn: () => {
   163→   163→      const payload = {
   164→   164→        name: form.name,
   165→   165→        description: form.description,
   166→   166→        price: Number(form.price.replace(',', '.')),
   167→   167→        prepTime: Number(form.prepTime),
   168→   168→        emoji: form.emoji || '🍽️',
   169→   169→        image: form.image || null,
   170→   170→        categoryId: form.categoryId,
   171→   171→      }
   172→   172→      return product ? apiPatch(`/api/products/${product.id}`, payload) : apiPost('/api/products', payload)
   173→   173→    },
   174→   174→    onSuccess: () => {
   175→   175→      void qc.invalidateQueries({ queryKey: ['categories'] })
   176→   176→      void qc.invalidateQueries({ queryKey: ['products'] })
   177→   177→      toast.success(product ? 'Produto atualizado' : 'Produto criado')
   178→   178→      onClose()
   179→   179→    },
   180→   180→    onError: (e: Error) => toast.error(e.message),
   181→   181→  })
   182→   182→
   183→   183→  return (
   184→   184→    <Dialog open={open} onOpenChange={(v) => !v && onClose()}>
   185→   185→      <DialogContent className="max-w-md max-h-[90vh] overflow-y-auto">
   186→   186→        <DialogHeader>
   187→   187→          <DialogTitle>{product ? 'Editar produto' : 'Novo produto'}</DialogTitle>
   188→   188→        </DialogHeader>
   189→   189→        <div className="space-y-3.5">
   190→   190→          <div className="grid grid-cols-2 gap-3">
   191→   191→            <div className="space-y-1.5">
   192→   192→              <Label>Nome</Label>
   193→   193→              <Input value={form.name} onChange={(e) => setForm({ ...form, name: e.target.value })} placeholder="Ex: Pizza Margherita" />
   194→   194→            </div>
   195→   195→            <div className="space-y-1.5">
   196→   196→              <Label>Categoria</Label>
   197→   197→              <Select value={form.categoryId} onValueChange={(v) => setForm({ ...form, categoryId: v })}>
   198→   198→                <SelectTrigger><SelectValue placeholder="Selecione" /></SelectTrigger>
   199→   199→                <SelectContent>
   200→   200→                  {(data?.categories ?? []).map((c) => (
   201→   201→                    <SelectItem key={c.id} value={c.id}>{c.icon} {c.name}</SelectItem>
   202→   202→                  ))}
   203→   203→                </SelectContent>
   204→   204→              </Select>
   205→   205→            </div>
   206→   206→          </div>
   207→   207→          <div className="space-y-1.5">
   208→   208→            <Label>Descrição</Label>
   209→   209→            <Textarea value={form.description} onChange={(e) => setForm({ ...form, description: e.target.value })} className="min-h-[60px] text-sm" placeholder="Ingredientes e diferenciais" />
   210→   210→          </div>
   211→   211→          <div className="grid grid-cols-3 gap-3">
   212→   212→            <div className="space-y-1.5">
   213→   213→              <Label>Preço (R$)</Label>
   214→   214→              <Input value={form.price} onChange={(e) => setForm({ ...form, price: e.target.value })} placeholder="49.90" inputMode="decimal" />
   215→   215→            </div>
   216→   216→            <div className="space-y-1.5">
   217→   217→              <Label>Preparo (min)</Label>
   218→   218→              <Input value={form.prepTime} onChange={(e) => setForm({ ...form, prepTime: e.target.value.replace(/\D/g, '') })} inputMode="numeric" />
   219→   219→            </div>
   220→   220→            <div className="space-y-1.5">
   221→   221→              <Label>Emoji</Label>
   222→   222→              <Input value={form.emoji} onChange={(e) => setForm({ ...form, emoji: e.target.value.slice(0, 2) })} />
   223→   223→            </div>
   224→   224→          </div>
   225→   225→          <div className="space-y-1.5">
   226→   226→            <Label className="flex items-center gap-1.5"><ImageIcon className="h-3.5 w-3.5" /> URL da imagem (opcional)</Label>
   227→   227→            <Input value={form.image} onChange={(e) => setForm({ ...form, image: e.target.value })} placeholder="https://…" />
   228→   228→          </div>
   229→   229→        </div>
   230→   230→        <DialogFooter>
   231→   231→          <Button className="w-full apex-gradient text-white font-semibold" onClick={() => save.mutate()} disabled={save.isPending || !form.name || !form.categoryId}>
   232→   232→            {save.isPending && <Loader2 className="h-4 w-4 animate-spin" />} Salvar produto
   233→   233→          </Button>
   234→   234→        </DialogFooter>
   235→   235→      </DialogContent>
   236→   236→    </Dialog>
   237→   237→  )
   238→   238→}
